String twoSum(List<int> nums, int target) {
  final seen = <int, int>{};
  for (var i = 0; i < nums.length; i++) {
    final j = seen[target - nums[i]];
    if (j != null) return '$j $i';
    seen.putIfAbsent(nums[i], () => i);
  }
  return 'none';
}

  print(twoSum([3, 2, 4], 6));
  print(twoSum([3, 3], 6));
  print(twoSum([1, 2, 3], 7));
}

const values = {'I': 1, 'V': 5, 'X': 10, 'L': 50, 'C': 100, 'D': 500, 'M': 1000};

int romanToInt(String s) {
  var total = 0;
  for (var i = 0; i < s.length; i++) {
    final v = values[s[i]]!;
    if (i + 1 < s.length && values[s[i + 1]]! > v) {
      total -= v;
    } else {
      total += v;
    }
  }
  return total;
}
--- main
void main() {
  for (final s in ['MCMXCIV', 'XLII', 'III', 'CDXLIV', 'MMXXVI']) {
    print('$s = ${romanToInt(s)}');
  }
}

List<int> spiral(List<List<int>> m) {
  final out = <int>[];
  if (m.isEmpty) return out;
  var top = 0, bottom = m.length - 1, left = 0, right = m[0].length - 1;
  while (top <= bottom && left <= right) {
    for (var c = left; c <= right; c++) out.add(m[top][c]);
    top++;
    for (var r = top; r <= bottom; r++) out.add(m[r][right]);
    right--;
    if (top <= bottom) {
      for (var c = right; c >= left; c--) out.add(m[bottom][c]);
      bottom--;
    }
    if (left <= right) {
      for (var r = bottom; r >= top; r--) out.add(m[r][left]);
      left++;
    }
  }
  return out;
}
--- main
List<List<int>> matrix(int rows, int cols) =>
    List.generate(rows, (r) => List.generate(cols, (c) => r * cols + c + 1));

void main() {
  for (final d in [[3, 4], [4, 4], [1, 3], [3, 1]]) {
    print(spiral(matrix(d[0], d[1])).join(' '));
  }
}
