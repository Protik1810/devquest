--- sol
primes_up_to <- function(n) {
  if (n < 2) return(integer(0))
  is_prime <- rep(TRUE, n)
  is_prime[1] <- FALSE
  p <- 2
  while (p * p <= n) {
    if (is_prime[p]) is_prime[seq(p * p, n, by = p)] <- FALSE
    p <- p + 1
  }
  which(is_prime)
}
--- main

cat("count(100): ", length(primes_up_to(100)), "\n", sep = "")

--- sol
top_words <- function(text, n) {
  words <- strsplit(tolower(text), "[^a-z]+")[[1]]
  words <- words[words != ""]
  counts <- table(words)
  counts <- counts[order(-counts, names(counts))]
  top <- head(counts, n)
  paste0(names(top), ": ", as.integer(top))
}
--- main
for (text in c("the cat and the hat and the bat sat on the mat", "Go go GO! Rust, rust; C.")) {
  writeLines(top_words(text, 3))
}

spiral <- function(m) {
  out <- integer(0)
  while (length(m) > 0) {
    out <- c(out, m[1, ])
    m <- m[-1, , drop = FALSE]
    if (nrow(m) == 0) break
    m <- t(m)[ncol(m):1, , drop = FALSE]
  }
  out
}
--- main
for (d in list(c(3, 4), c(4, 4), c(1, 3), c(3, 1))) {
  m <- matrix(seq_len(d[1] * d[2]), nrow = d[1], ncol = d[2], byrow = TRUE)
  cat(paste(spiral(m), collapse = " "), "\n", sep = "")
}
