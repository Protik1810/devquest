### caesar | Easy
fn: `function caesar(string $text, int $k): string`
hint: `ord()` and `chr()` convert between characters and their codes; `str_split()` gives you the characters.
--- head

--- stub

--- sol
function caesar(string $text, int $k): string {
    $shift = (($k % 26) + 26) % 26;
    $out = '';
    foreach (str_split($text) as $c) {
        if ($c >= 'A' && $c <= 'Z') {
            $out .= chr((ord($c) - 65 + $shift) % 26 + 65);
        } elseif ($c >= 'a' && $c <= 'z') {
            $out .= chr((ord($c) - 97 + $shift) % 26 + 97);
        } else {
            $out .= $c;
        }
    }
    return $out;
}
--- main
echo caesar('Hello, World!', 3), "\n";
echo caesar('xyz ABC', 2), "\n";
echo caesar('Khoor, Zruog!', -3), "\n";
echo caesar('DevQuest', 13), "\n";

### roman | Medium
fn: `function romanToInt(string $s): int`
hint: An associative array `['I' => 1, ...]` is the lookup table; `$s[$i]` reads one character.
--- head

--- stub

    // TODO

}
--- sol
function romanToInt(string $s): int {
    $values = ['I' => 1, 'V' => 5, 'X' => 10, 'L' => 50, 'C' => 100, 'D' => 500, 'M' => 1000];
    $total = 0;
    for ($i = 0; $i < strlen($s); $i++) {
        $v = $values[$s[$i]];
        if ($i + 1 < strlen($s) && $values[$s[$i + 1]] > $v) {
            $total -= $v;
        } else {
            $total += $v;
        }
    }
    return $total;
}
--- main
foreach (['MCMXCIV', 'XLII', 'III', 'CDXLIV', 'MMXXVI'] as $s) {
    echo "$s = " . romanToInt($s) . "\n";
}

### anagrams | Hard
fn: `function groupAnagrams(array $words): array`
hint: PHP arrays remember insertion order, so `$groups[$key][] = $w;` followed by `array_values()` is enough.

--- sol
function groupAnagrams(array $words): array {
    $groups = [];
    foreach ($words as $w) {
        $letters = str_split($w);
        sort($letters);
        $groups[implode('', $letters)][] = $w;
    }
    return array_values($groups);
}
--- main
$inputs = [
    ['eat', 'tea', 'tan', 'ate', 'nat', 'bat'],
    ['listen', 'silent', 'enlist', 'google', 'gogole', 'cat', 'act'],
];
foreach ($inputs as $words) {
    foreach (groupAnagrams($words) as $group) {
        echo implode(' ', $group), "\n";
    }
}
