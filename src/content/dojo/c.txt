/* Shift every letter of text by k places, writing the result into out. */
void caesar(const char *text, int k, char *out) {
    int shift = ((k % 26) + 26) % 26;
    size_t i;
    for (i = 0; text[i]; i++) {
        char c = text[i];
        if (c >= 'A' && c <= 'Z') c = (char)('A' + (c - 'A' + shift) % 26);
        else if (c >= 'a' && c <= 'z') c = (char)('a' + (c - 'a' + shift) % 26);
        out[i] = c;
    }
    out[i] = '\0';
}

/* Return the index of target in the sorted array a of length n, or -1. */
int binary_search(const int *a, int n, int target) {
    int lo = 0, hi = n - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;
        if (a[mid] == target) return mid;
        if (a[mid] < target) lo = mid + 1;
        else hi = mid - 1;
    }
    return -1;
}

        printf("search(%d) = %d\n", targets[i], binary_search(a, 7, targets[i]));
    }

}

/* Write the spiral order of the rows x cols matrix m into out; return the number of values written. */
int spiral(const int *m, int rows, int cols, int *out) {
    int top = 0, bottom = rows - 1, left = 0, right = cols - 1, n = 0;
    while (top <= bottom && left <= right) {
        for (int c = left; c <= right; c++) out[n++] = m[top * cols + c];
        top++;
        for (int r = top; r <= bottom; r++) out[n++] = m[r * cols + right];
        right--;
        if (top <= bottom) {
            for (int c = right; c >= left; c--) out[n++] = m[bottom * cols + c];
            bottom--;
        }
        if (left <= right) {
            for (int r = bottom; r >= top; r--) out[n++] = m[r * cols + left];
            left++;
        }
    }
    return n;
}
--- main
int main(void) {
    int dims[4][2] = {{3, 4}, {4, 4}, {1, 3}, {3, 1}};
    int m[16], out[16];
    for (int d = 0; d < 4; d++) {
        int rows = dims[d][0], cols = dims[d][1];
        for (int i = 0; i < rows * cols; i++) m[i] = i + 1;
        int n = spiral(m, rows, cols, out);
        for (int i = 0; i < n; i++) printf(i ? " %d" : "%d", out[i]);
        printf("\n");
    }
    return 0;
}
