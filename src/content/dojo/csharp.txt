string TwoSum(int[] nums, int target)
{
    var seen = new Dictionary<int, int>();
    for (int i = 0; i < nums.Length; i++)
    {
        if (seen.TryGetValue(target - nums[i], out int j)) return $"{j} {i}";
        seen.TryAdd(nums[i], i);
    }
    return "none";
}

--- sol
string[] TopWords(string text, int n) =>
    Regex.Split(text.ToLowerInvariant(), "[^a-z]+")
        .Where(w => w.Length > 0)
        .GroupBy(w => w)
        .OrderByDescending(g => g.Count())
        .ThenBy(g => g.Key, StringComparer.Ordinal)
        .Take(n)
        .Select(g => $"{g.Key}: {g.Count()}")
        .ToArray();
--- main

    foreach (var line in TopWords(text, 3))
        Console.WriteLine(line);

List<int[]> Merge(int[][] intervals)
{
    var result = new List<int[]>();
    foreach (var iv in intervals.OrderBy(i => i[0]))
    {
        if (result.Count > 0 && iv[0] <= result[^1][1])
            result[^1][1] = Math.Max(result[^1][1], iv[1]);
        else
            result.Add(new[] { iv[0], iv[1] });
    }
    return result;
}

    new[] { new[] { 5, 7 }, new[] { 1, 2 }, new[] { 2, 3 } },
