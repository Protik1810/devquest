def rle(s)
  s.chars.chunk_while { |a, b| a == b }.map { |run| "#{run[0]}#{run.size}" }.join
end
--- main

  puts "#{s} -> #{rle(s)}"
end

end
--- sol
def top_words(text, n)
  text.downcase.scan(/[a-z]+/).tally
      .sort_by { |word, count| [-count, word] }
      .first(n)
      .map { |word, count| "#{word}: #{count}" }
end
--- main
["the cat and the hat and the bat sat on the mat", "Go go GO! Rust, rust; C."].each do |text|
  puts top_words(text, 3)
end

def longest_unique(s)
  last = {}
  start = best = best_start = 0
  s.each_char.with_index do |c, i|
    start = last[c] + 1 if last[c] && last[c] >= start
    last[c] = i
    if i - start + 1 > best
      best = i - start + 1
      best_start = start
    end
  end
  s[best_start, best]
end
--- main
%w[abcabcbb bbbbb pwwkew dvdf devquest].each do |s|
  sub = longest_unique(s)
  puts "#{s} -> #{sub.length} #{sub}"
end
