=== module Capstone projects

=== lesson java-project | Capstone 1: a library management system | 400
This capstone builds the core of a library system in three steps, the way a feature grows in a real codebase: first a **domain model**, then **business rules** with exceptions and dates, then **reporting** with streams and file import. Each step's starter contains the previous step's solution, so you can continue even if a step gave you trouble.

## How a task like this is approached

1. **Model the domain** before writing logic: what are the things (books, members, loans), which are values (records) and which hold changing state (the library)?
2. **Protect invariants in one place**: a book can't have an invalid year, a copy can't be lent twice, a member can't exceed the loan limit. Validation lives in constructors and in the methods that change state, and violations become exceptions with clear messages.
3. **Keep the API small**: the `main` method (your "client") only uses a handful of public methods; the maps behind them are private.
4. **Report with streams**: once the data is right, questions are short pipelines.

Each step's `main` method is the specification: it calls your classes and its printed output must match exactly. Don't change `main`.
--- goals
- design a domain model with records, a stateful class and custom exceptions
- enforce business rules and compute fines with java.time and BigDecimal
- import data from a CSV file, rejecting invalid lines
- write reports with streams and collectors

record Book(String isbn, String title, String author, int year) {
    Book {
        if (isbn == null || isbn.isBlank()) throw new IllegalArgumentException("ISBN is required");
        if (title == null || title.isBlank()) throw new IllegalArgumentException("title is required");
        if (year < 1450 || year > 2100) throw new IllegalArgumentException("invalid year " + year);
    }

    @Override
    public String toString() {
        return title + " (" + author + ", " + year + ")";
    }
}

class Library {
    private final Map<String, Book> books = new HashMap<>();

    boolean addBook(Book book) {
        return books.putIfAbsent(book.isbn(), book) == null;
    }

    int size() {
        return books.size();
    }

    Optional<Book> findByIsbn(String isbn) {
        return Optional.ofNullable(books.get(isbn));
    }

    List<Book> search(String text) {
        String needle = text.toLowerCase();
        return books.values().stream()
            .filter(b -> b.title().toLowerCase().contains(needle) || b.author().toLowerCase().contains(needle))
            .sorted(Comparator.comparing(Book::title))
            .toList();
    }

    List<String> titlesBefore(int year) {
        return books.values().stream()
            .filter(b -> b.year() < year)
            .sorted(Comparator.comparing(Book::year).thenComparing(Book::title))
            .map(Book::title)
            .toList();
    }
}

class LibraryException extends RuntimeException {
    LibraryException(String message) {
        super(message);
    }
}

class Library {
    static final int MAX_LOANS = 3;
    static final int LOAN_DAYS = 14;
    static final BigDecimal FINE_PER_DAY = new BigDecimal("0.25");

    private final Map<String, Book> books = new HashMap<>();
    private final Map<Integer, Member> members = new HashMap<>();
    private final Map<String, Loan> activeLoans = new HashMap<>();

    boolean addBook(Book book) {
        return books.putIfAbsent(book.isbn(), book) == null;
    }

    void addMember(Member member) {
        members.put(member.id(), member);
    }

    int size() {
        return books.size();
    }

    Optional<Book> findByIsbn(String isbn) {
        return Optional.ofNullable(books.get(isbn));
    }

    List<Book> search(String text) {
        String needle = text.toLowerCase();
        return books.values().stream()
            .filter(b -> b.title().toLowerCase().contains(needle) || b.author().toLowerCase().contains(needle))
            .sorted(Comparator.comparing(Book::title))
            .toList();
    }

    List<String> titlesBefore(int year) {
        return books.values().stream()
            .filter(b -> b.year() < year)
            .sorted(Comparator.comparing(Book::year).thenComparing(Book::title))
            .map(Book::title)
            .toList();
    }

    Loan borrow(String isbn, int memberId, LocalDate date) {
        Book book = findByIsbn(isbn).orElseThrow(() -> new LibraryException("unknown book " + isbn));
        Member member = members.get(memberId);
        if (member == null) throw new LibraryException("unknown member " + memberId);
        if (activeLoans.containsKey(isbn)) throw new LibraryException(book.title() + " is already on loan");
        long count = activeLoans.values().stream().filter(l -> l.member().equals(member)).count();
        if (count >= MAX_LOANS) throw new LibraryException(member.name() + " already has " + MAX_LOANS + " loans");
        Loan loan = new Loan(book, member, date, date.plusDays(LOAN_DAYS));
        activeLoans.put(isbn, loan);
        return loan;
    }

    BigDecimal giveBack(String isbn, LocalDate date) {
        Loan loan = activeLoans.remove(isbn);
        if (loan == null) throw new LibraryException(isbn + " is not on loan");
        long late = Math.max(0, ChronoUnit.DAYS.between(loan.dueOn(), date));
        return FINE_PER_DAY.multiply(BigDecimal.valueOf(late));
    }

    List<Loan> loansOf(int memberId) {
        return activeLoans.values().stream()
            .filter(l -> l.member().id() == memberId)
            .sorted(Comparator.comparing(Loan::dueOn).thenComparing(l -> l.book().title()))
            .toList();
    }
}

class Library {
    static final int MAX_LOANS = 3;
    static final int LOAN_DAYS = 14;
    static final BigDecimal FINE_PER_DAY = new BigDecimal("0.25");

    private final Map<String, Book> books = new HashMap<>();
    private final Map<Integer, Member> members = new HashMap<>();
    private final Map<String, Loan> activeLoans = new HashMap<>();
    private final List<Loan> history = new ArrayList<>();
    private BigDecimal finesCollected = BigDecimal.ZERO;

    boolean addBook(Book book) {
        return books.putIfAbsent(book.isbn(), book) == null;
    }

    void addMember(Member member) {
        members.put(member.id(), member);
    }

    int size() {
        return books.size();
    }

    Optional<Book> findByIsbn(String isbn) {
        return Optional.ofNullable(books.get(isbn));
    }

    List<Book> search(String text) {
        String needle = text.toLowerCase();
        return books.values().stream()
            .filter(b -> b.title().toLowerCase().contains(needle) || b.author().toLowerCase().contains(needle))
            .sorted(Comparator.comparing(Book::title))
            .toList();
    }

    List<String> titlesBefore(int year) {
        return books.values().stream()
            .filter(b -> b.year() < year)
            .sorted(Comparator.comparing(Book::year).thenComparing(Book::title))
            .map(Book::title)
            .toList();
    }

    Loan borrow(String isbn, int memberId, LocalDate date) {
        Book book = findByIsbn(isbn).orElseThrow(() -> new LibraryException("unknown book " + isbn));
        Member member = members.get(memberId);
        if (member == null) throw new LibraryException("unknown member " + memberId);
        if (activeLoans.containsKey(isbn)) throw new LibraryException(book.title() + " is already on loan");
        long count = activeLoans.values().stream().filter(l -> l.member().equals(member)).count();
        if (count >= MAX_LOANS) throw new LibraryException(member.name() + " already has " + MAX_LOANS + " loans");
        Loan loan = new Loan(book, member, date, date.plusDays(LOAN_DAYS));
        activeLoans.put(isbn, loan);
        history.add(loan);
        return loan;
    }

    BigDecimal giveBack(String isbn, LocalDate date) {
        Loan loan = activeLoans.remove(isbn);
        if (loan == null) throw new LibraryException(isbn + " is not on loan");
        long late = Math.max(0, ChronoUnit.DAYS.between(loan.dueOn(), date));
        BigDecimal fine = FINE_PER_DAY.multiply(BigDecimal.valueOf(late));
        finesCollected = finesCollected.add(fine);
        return fine;
    }

    List<Loan> loansOf(int memberId) {
        return activeLoans.values().stream()
            .filter(l -> l.member().id() == memberId)
            .sorted(Comparator.comparing(Loan::dueOn).thenComparing(l -> l.book().title()))
            .toList();
    }

    /** Adds the books in a "isbn;title;author;year" file (header first); returns how many lines were rejected. */
    int importCsv(Path file) throws IOException {
        int rejected = 0;
        List<String> lines = Files.readAllLines(file);
        for (String line : lines.subList(1, lines.size())) {
            if (line.isBlank()) continue;
            String[] parts = line.split(";");
            try {
                if (parts.length != 4) throw new IllegalArgumentException("expected 4 fields");
                if (!addBook(new Book(parts[0].strip(), parts[1].strip(), parts[2].strip(), Integer.parseInt(parts[3].strip())))) rejected++;
            } catch (IllegalArgumentException e) {
                rejected++;
            }
        }
        return rejected;
    }

    List<String> mostBorrowed(int n) {
        Map<String, Long> counts = history.stream().collect(Collectors.groupingBy(l -> l.book().title(), Collectors.counting()));
        return counts.entrySet().stream()
            .sorted(Comparator.comparing((Map.Entry<String, Long> e) -> -e.getValue()).thenComparing(Map.Entry::getKey))
            .limit(n)
            .map(e -> e.getKey() + ": " + e.getValue())
            .toList();
    }

    List<String> overdue(LocalDate today) {
        return activeLoans.values().stream()
            .filter(l -> l.dueOn().isBefore(today))
            .sorted(Comparator.comparing(Loan::dueOn).thenComparing(l -> l.book().title()))
            .map(l -> l.book().title() + " - " + l.member().name() + " (" + ChronoUnit.DAYS.between(l.dueOn(), today) + " days)")
            .toList();
    }

    BigDecimal finesCollected() {
        return finesCollected;
    }
}

=== lesson java-capstone-bank | Capstone 2: a bank that survives concurrency | 450
Money must never appear or disappear. In this capstone you build a small bank whose transfers are correct first in one thread, then under heavy **concurrency**: twenty thousand transfers running at the same time on virtual threads. It's the kind of problem behind payment systems, inventory counts and ticket sales.

## How a task like this is approached

1. **Get it right single-threaded**: clear rules, validated inputs, exceptions with useful messages, integer cents instead of floating-point money.
2. **Find the shared mutable state**: here, account balances and the transfer history.
3. **Protect it**: each account gets its own lock; a transfer needs **two** locks, and taking them in different orders in different threads causes **deadlocks**, so locks are always taken in a fixed global order (by account id).
4. **Check invariants after the storm**: the total amount of money must be exactly what was deposited, no balance may be negative, and every transfer must be either recorded or refused.
--- goals
- implement business rules with exceptions and integer money
- identify shared mutable state and protect it with locks
- avoid deadlocks with a global lock order
- verify concurrent code with invariants

**Step 2: make it thread-safe.** The new `main` opens 50 accounts and runs **20,000 random transfers concurrently** on virtual threads. Run the starter first: the unprotected bank from step 1 loses (or creates) money, because `balance -= cents` races between threads.

Make the bank safe:

- accounts in a `ConcurrentHashMap`, the history in a thread-safe collection (`ConcurrentLinkedQueue`);
- each `Account` has its own `ReentrantLock`; `transfer` locks **both** accounts, always the **lower id first** (otherwise two opposite transfers can deadlock), checks the funds and moves the money while holding the locks, and releases them in `finally`;
- `balance(id)` reads under the account's lock;
- keep the opening total (an `AtomicLong`), and write `audit()`, returning `audit ok` when the total equals the opening total and no balance is negative, else `audit FAILED`.

class Account {
    final int id;
    final String owner;
    final ReentrantLock lock = new ReentrantLock();
    long balanceCents;              // guarded by lock

    Account(int id, String owner, long balanceCents) {
        this.id = id;
        this.owner = owner;
        this.balanceCents = balanceCents;
    }
}

class Bank {
    private final Map<Integer, Account> accounts = new ConcurrentHashMap<>();
    private final Queue<Transfer> history = new ConcurrentLinkedQueue<>();
    private final AtomicLong openingTotal = new AtomicLong();

    void open(int id, String owner, long cents) {
        if (accounts.putIfAbsent(id, new Account(id, owner, cents)) != null) throw new BankException("account " + id + " already exists");
        openingTotal.addAndGet(cents);
    }

    private Account get(int id) {
        Account a = accounts.get(id);
        if (a == null) throw new BankException("unknown account " + id);
        return a;
    }

    void transfer(int fromId, int toId, long cents) {
        if (cents <= 0) throw new BankException("amount must be positive");
        if (fromId == toId) throw new BankException("cannot transfer to the same account");
        Account from = get(fromId);
        Account to = get(toId);
        // Always lock the lower id first: two opposite transfers can then never wait for each other (no deadlock)
        Account first = fromId < toId ? from : to;
        Account second = fromId < toId ? to : from;
        first.lock.lock();
        try {
            second.lock.lock();
            try {
                if (from.balanceCents < cents) throw new BankException(from.owner + " has insufficient funds");
                from.balanceCents -= cents;
                to.balanceCents += cents;
                history.add(new Transfer(fromId, toId, cents));
            } finally {
                second.lock.unlock();
            }
        } finally {
            first.lock.unlock();
        }
    }

    long balance(int id) {
        Account a = get(id);
        a.lock.lock();
        try {
            return a.balanceCents;
        } finally {
            a.lock.unlock();
        }
    }

    long total() {
        return accounts.keySet().stream().mapToLong(this::balance).sum();
    }

    int transfersCompleted() {
        return history.size();
    }

    String audit() {
        boolean negative = accounts.keySet().stream().anyMatch(id -> balance(id) < 0);
        return total() == openingTotal.get() && !negative ? "audit ok" : "audit FAILED";
    }

    static String format(long cents) {
        return String.format("%d.%02d", cents / 100, cents % 100);
    }
}

::hints
- `Account first = fromId < toId ? from : to;` then lock `first`, then `second`, each with `try { ... } finally { unlock(); }`.

=== lesson java-capstone-solo | Capstone 3: build it from a spec | 450
The final capstone gives you a **specification** and a test harness, and nothing else: you design the types and write them from scratch. A classic object-oriented design exercise (and a popular interview question): a **vending machine**.

## The specification

**Types** (names and methods are fixed by the harness):

- `enum Coin` with the values `FIVE`, `TEN`, `TWENTY`, `FIFTY`, `ONE_EURO`, `TWO_EUROS` (worth 5, 10, 20, 50, 100 and 200 cents) and a method `int cents()`;
- `record Product(String code, String name, int priceCents)`;
- `record Purchase(Product product, List<Coin> change)`;
- `class VendingException extends RuntimeException`;
- `class VendingMachine` with the methods below, and a `static String euros(int cents)` that formats cents like `1.50`.

**Rules:**

1. `load(product, quantity)` adds stock for the product's code; loading a code again replaces the product's details (name, price) and adds to its stock.
2. `addChange(coin, count)` puts coins into the machine's coin store.
3. `insert(coin)` adds the coin to the current credit **and** to the coin store. `credit()` returns the current credit in cents.
4. `buy(code)` checks, in this order: unknown code → `unknown product <code>`; no stock → `<name> is sold out`; credit below the price → `insert <missing> more` (formatted like `0.50`); change impossible → `cannot make change`. A refused purchase changes nothing (the credit stays).
5. Change is computed **greedily**: repeatedly take the largest coin that fits the remaining amount and is still in the coin store. If that doesn't reach exactly zero, the change is impossible.
6. A successful purchase removes the change coins from the store, reduces the stock by one, records the sale, sets the credit to zero, and returns the product and the change (largest coins first).
7. `cancel()` returns the coins inserted since the last purchase, in insertion order, removes them from the coin store and sets the credit to zero.
8. `salesReport()` returns one line per product sold, by code, as `A1 Cola x2 = 3.00` (using the product's current name and the total of its sales), then `total 4.90`; every line ends with a line break.

## How to approach it

1. Write the types and method signatures first, with placeholder bodies, so the harness compiles.
2. Choose the state: maps from code to product, to stock and to sales; an `EnumMap<Coin, Integer>` for the coin store; a list of inserted coins.
3. Write the change-making as its own private method that works on a **copy** of the coin store, so a failed attempt changes nothing.
4. Run the harness and compare its output line by line with the expected output.
--- goals
- design a set of types from a written specification
- choose data structures for state: maps, EnumMap and lists
- implement an algorithm (greedy change-making) that fails without side effects
- keep refused operations free of partial changes

enum Coin {
    FIVE(5), TEN(10), TWENTY(20), FIFTY(50), ONE_EURO(100), TWO_EUROS(200);

    private final int cents;

    Coin(int cents) {
        this.cents = cents;
    }

    int cents() {
        return cents;
    }
}

record Product(String code, String name, int priceCents) {}

record Purchase(Product product, List<Coin> change) {}

class VendingException extends RuntimeException {
    VendingException(String message) {
        super(message);
    }
}

class VendingMachine {
    private final Map<String, Product> products = new TreeMap<>();
    private final Map<String, Integer> stock = new TreeMap<>();
    private final Map<String, Integer> sold = new TreeMap<>();
    private final Map<String, Integer> revenue = new TreeMap<>();
    private final Map<Coin, Integer> coins = new EnumMap<>(Coin.class);
    private final List<Coin> inserted = new ArrayList<>();

    static String euros(int cents) {
        return String.format("%d.%02d", cents / 100, cents % 100);
    }

    void load(Product product, int quantity) {
        products.put(product.code(), product);
        stock.merge(product.code(), quantity, Integer::sum);
    }

    void addChange(Coin coin, int count) {
        coins.merge(coin, count, Integer::sum);
    }

    void insert(Coin coin) {
        inserted.add(coin);
        coins.merge(coin, 1, Integer::sum);
    }

    int credit() {
        return inserted.stream().mapToInt(Coin::cents).sum();
    }

    Purchase buy(String code) {
        Product product = products.get(code);
        if (product == null) throw new VendingException("unknown product " + code);
        if (stock.getOrDefault(code, 0) == 0) throw new VendingException(product.name() + " is sold out");
        int credit = credit();
        if (credit < product.priceCents()) throw new VendingException("insert " + euros(product.priceCents() - credit) + " more");
        List<Coin> change = makeChange(credit - product.priceCents());
        if (change == null) throw new VendingException("cannot make change");
        for (Coin c : change) coins.merge(c, -1, Integer::sum);
        stock.merge(code, -1, Integer::sum);
        sold.merge(code, 1, Integer::sum);
        revenue.merge(code, product.priceCents(), Integer::sum);
        inserted.clear();
        return new Purchase(product, change);
    }

    /** Greedy: the largest coin that fits and is available, until nothing is left; null if impossible. */
    private List<Coin> makeChange(int amount) {
        Map<Coin, Integer> available = new EnumMap<>(coins);
        List<Coin> change = new ArrayList<>();
        Coin[] values = Coin.values();
        for (int i = values.length - 1; i >= 0 && amount > 0; i--) {
            Coin c = values[i];
            while (amount >= c.cents() && available.getOrDefault(c, 0) > 0) {
                amount -= c.cents();
                available.merge(c, -1, Integer::sum);
                change.add(c);
            }
        }
        return amount == 0 ? change : null;
    }

    List<Coin> cancel() {
        List<Coin> back = new ArrayList<>(inserted);
        for (Coin c : back) coins.merge(c, -1, Integer::sum);
        inserted.clear();
        return back;
    }

    String salesReport() {
        StringBuilder sb = new StringBuilder();
        int total = 0;
        for (Map.Entry<String, Integer> e : sold.entrySet()) {
            sb.append(e.getKey()).append(' ').append(products.get(e.getKey()).name())
                .append(" x").append(e.getValue()).append(" = ").append(euros(revenue.get(e.getKey()))).append('\n');
            total += revenue.get(e.getKey());
        }
        return sb.append("total ").append(euros(total)).append('\n').toString();
    }
}

::hints
- An `EnumMap<Coin, Integer>` holds coin counts; `merge(coin, 1, Integer::sum)` adds one.
- Greedy change: loop over `Coin.values()` from the largest, taking coins while they fit and are available in a copy of the store.
- Throw the exceptions before changing any state, so a refused purchase leaves everything as it was.
--- cards

Greedy algorithm :: Take the best local choice repeatedly; works for standard coin systems, can fail with limited coins.
