=== lesson py-boss | Capstone 1: a budget tracker | 400
Time to build something real from start to finish, the way professional developers do: **plan, build in small tested steps, then polish**. You'll write a budget tracker that records transactions, imports a bank's CSV export, produces a monthly report with budget alerts, and saves its data as JSON.

## How a task like this is approached

1. **Clarify the requirements.** Write them as user stories: *"As a user, I can import my bank's CSV export, so I don't type transactions by hand."* *"As a user, I see where my money went each month."* *"As a user, I'm warned when a category goes over budget."*
2. **Sketch the design.** Nouns and verbs give the classes: a `Transaction` (date, description, amount, category) and a `Ledger` that holds them and answers questions. Importing, reporting and saving are separate **functions** around the model, each with one responsibility, so the model never depends on file formats or screen layout.
3. **Decide the hard rules up front.** Money is stored as **integer cents**, never floats (Module 1!). Dates are real `date` objects, written as ISO 8601 text in files. Invalid data raises one domain exception, `LedgerError`.
4. **Build in vertical slices**, each fully working and tested before the next:
   - Step 1: the model (`Transaction`, `Ledger`);
   - Step 2: importing CSV, collecting errors instead of crashing on the first bad line;
   - Step 3: the monthly report and budget alerts;
   - Step 4: saving and loading JSON, with a round-trip test.
5. **Refactor as you go**, and keep each function small enough to test directly.

## A note on Decimal

Bank exports write amounts like `-12.50`. Converting with `float` risks errors like `0.1 + 0.2`. The standard `decimal` module represents decimal numbers **exactly**: `Decimal("-12.50") * 100` is exactly `-1250`. Step 2 uses it to turn text into whole cents safely.

Each step's starter contains a working solution to the previous steps, so you can always continue. When you're done, you'll have written about 150 lines of clean, tested, object-oriented Python, the size of a real small tool.

from dataclasses import dataclass
from datetime import date


class LedgerError(Exception):
    pass


@dataclass(frozen=True)
class Transaction:
    when: date
    description: str
    amount_cents: int
    category: str = "uncategorised"

    def __post_init__(self):
        if not self.description.strip():
            raise LedgerError("a transaction needs a description")
        if self.amount_cents == 0:
            raise LedgerError("a transaction can't be zero")


class Ledger:
    def __init__(self):
        self._transactions = []

    def add(self, transaction):
        self._transactions.append(transaction)

    def __len__(self):
        return len(self._transactions)

    def balance(self):
        return sum(t.amount_cents for t in self._transactions)

    def in_month(self, year, month):
        found = [t for t in self._transactions if t.when.year == year and t.when.month == month]
        return sorted(found, key=lambda t: t.when)

    def spending_by_category(self, year, month):
        totals = {}
        for t in self.in_month(year, month):
            if t.amount_cents < 0:
                totals[t.category] = totals.get(t.category, 0) - t.amount_cents
        return dict(sorted(totals.items(), key=lambda kv: (-kv[1], kv[0])))

import csv
import io
from dataclasses import dataclass
from datetime import date
from decimal import Decimal, InvalidOperation


class LedgerError(Exception):
    pass


@dataclass(frozen=True)
class Transaction:
    when: date
    description: str
    amount_cents: int
    category: str = "uncategorised"

    def __post_init__(self):
        if not self.description.strip():
            raise LedgerError("a transaction needs a description")
        if self.amount_cents == 0:
            raise LedgerError("a transaction can't be zero")


def to_cents(text):
    value = Decimal(text.strip()) * 100
    if value != value.to_integral_value():
        raise InvalidOperation(text)
    return int(value)


def parse_transactions(csv_text):
    transactions = []
    errors = []
    for line_no, row in enumerate(csv.DictReader(io.StringIO(csv_text)), start=2):
        try:
            when = date.fromisoformat(row["date"].strip())
        except ValueError:
            errors.append(f"line {line_no}: bad date {row['date']!r}")
            continue
        try:
            cents = to_cents(row["amount"])
        except InvalidOperation:
            errors.append(f"line {line_no}: bad amount {row['amount']!r}")
            continue
        category = row["category"].strip() or "uncategorised"
        try:
            transactions.append(Transaction(when, row["description"], cents, category))
        except LedgerError as e:
            errors.append(f"line {line_no}: {e}")
    return transactions, errors

class Ledger:
    def __init__(self):
        self._transactions = []

    def add(self, transaction):
        self._transactions.append(transaction)

    def __len__(self):
        return len(self._transactions)

    def balance(self):
        return sum(t.amount_cents for t in self._transactions)

    def in_month(self, year, month):
        found = [t for t in self._transactions if t.when.year == year and t.when.month == month]
        return sorted(found, key=lambda t: t.when)

    def spending_by_category(self, year, month):
        totals = {}
        for t in self.in_month(year, month):
            if t.amount_cents < 0:
                totals[t.category] = totals.get(t.category, 0) - t.amount_cents
        return dict(sorted(totals.items(), key=lambda kv: (-kv[1], kv[0])))


def money(cents):
    sign = "-" if cents < 0 else ""
    return f"{sign}{abs(cents) / 100:,.2f}"


def monthly_report(ledger, year, month):
    items = ledger.in_month(year, month)
    income = sum(t.amount_cents for t in items if t.amount_cents > 0)
    spending = sum(t.amount_cents for t in items if t.amount_cents < 0)
    lines = [date(year, month, 1).strftime("%B %Y")]
    for label, value in [("Income", income), ("Spending", spending), ("Net", income + spending)]:
        lines.append(f"{label:<10}{money(value):>12}")
    lines.append("Top spending:")
    for category, cents in list(ledger.spending_by_category(year, month).items())[:3]:
        lines.append(f"  {category:<14}{money(cents):>10}")
    return "\n".join(lines)


def check_budgets(ledger, year, month, budgets, notify):
    spent = ledger.spending_by_category(year, month)
    for category in sorted(budgets):
        used = spent.get(category, 0)
        if used > budgets[category]:
            notify(f"{category}: {used * 100 // budgets[category]}% of budget")

import json
from dataclasses import asdict, dataclass
from datetime import date


class LedgerError(Exception):
    pass


@dataclass(frozen=True)
class Transaction:
    when: date
    description: str
    amount_cents: int
    category: str = "uncategorised"

    def __post_init__(self):
        if not self.description.strip():
            raise LedgerError("a transaction needs a description")
        if self.amount_cents == 0:
            raise LedgerError("a transaction can't be zero")


class Ledger:
    def __init__(self):
        self._transactions = []

    def add(self, transaction):
        self._transactions.append(transaction)

    def __iter__(self):
        return iter(self._transactions)

    def __len__(self):
        return len(self._transactions)


def save_ledger(ledger, path):
    rows = []
    for t in ledger:
        row = asdict(t)
        row["when"] = t.when.isoformat()
        rows.append(row)
    with open(path, "w", encoding="utf-8") as f:
        json.dump({"version": 1, "transactions": rows}, f, indent=2)


def load_ledger(path):
    ledger = Ledger()
    try:
        with open(path, encoding="utf-8") as f:
            data = json.load(f)
    except FileNotFoundError:
        return ledger
    except json.JSONDecodeError as e:
        raise LedgerError(f"corrupt ledger file: {e}") from e
    if data.get("version") != 1:
        raise LedgerError(f"unsupported ledger version: {data.get('version')!r}")
    for row in data["transactions"]:
        ledger.add(Transaction(date.fromisoformat(row["when"]), row["description"], row["amount_cents"], row["category"]))
    return ledger

=== lesson py-capstone-cli | Capstone 2: a command-line text analyser | 400
Your second project is a **command-line tool**, the kind developers write every week to automate a job. `textstats` will read a text file and report its word count, vocabulary, longest words and most common words, with options, proper exit codes and JSON output for other programs.

## The plan

- **Step 1: tokenising.** Turn raw text into words with a regular expression, as a **generator** so huge files stream through without filling memory.
- **Step 2: analysing.** Compute the statistics with `Counter` and return them as a plain dictionary, with no printing, so the logic is easy to test and reuse.
- **Step 3: the command-line interface.** Parse arguments with `argparse`, read the file, print a readable report or JSON, write errors to **standard error**, and return an **exit code** (0 for success, non-zero for failure), as every well-behaved command-line program does.

This split is the classic **functional core, imperative shell** design: pure, testable functions in the middle (steps 1 and 2), and a thin layer that deals with the messy outside world (step 3).

## Command-line programs in Python

`sys.argv` holds the words typed after the program name. The standard `argparse` module turns them into a tidy object, generates `--help` automatically, and reports mistakes:

```python
import argparse

parser = argparse.ArgumentParser(prog="textstats", description="Word statistics for a text file")
parser.add_argument("path", help="the file to analyse")
parser.add_argument("--top", type=int, default=5, help="how many common words to show")
parser.add_argument("--json", action="store_true", help="print JSON instead of text")
args = parser.parse_args(["notes.txt", "--top", "3"])     # normally parse_args() reads sys.argv
args.path, args.top, args.json                            # 'notes.txt' 3 False
```

Passing the argument list explicitly (`main(argv)`) makes the whole program testable: tests call `main(["file.txt", "--json"])` and check what it prints and returns. To check printed output, tests capture it with `contextlib.redirect_stdout(io.StringIO())`.

Conventions worth following: normal output goes to **stdout**, errors and warnings to **stderr** (`print(..., file=sys.stderr)`), and the program ends with `sys.exit(main(sys.argv[1:]))` so the shell sees the exit code.

📚 **Go deeper:** Python "Argparse Tutorial" (docs.python.org/3/howto/argparse.html); Gary Bernhardt, "Functional Core, Imperative Shell"; *The Art of Unix Programming* (rules of modularity and silence).

import re

WORD = re.compile(r"[a-z]+(?:'[a-z]+)*")


def words(text):
    for match in WORD.finditer(text.lower()):
        yield match.group()


def sentence_count(text):
    parts = [p for p in re.split(r"[.!?]+", text) if p.strip()]
    return len(parts)

import argparse
import json
import re
import sys
from collections import Counter

STOPWORDS = {"the", "a", "an", "and", "or", "of", "to", "in", "is", "it"}
WORD = re.compile(r"[a-z]+(?:'[a-z]+)*")


def words(text):
    for match in WORD.finditer(text.lower()):
        yield match.group()


def sentence_count(text):
    parts = [p for p in re.split(r"[.!?]+", text) if p.strip()]
    return len(parts)


def analyse(text, top=5, stopwords=()):
    counts = Counter(words(text))
    total = sum(counts.values())
    letters = sum(len(w) * n for w, n in counts.items())
    ignored = set(stopwords)
    common = sorted(((w, n) for w, n in counts.items() if w not in ignored), key=lambda wn: (-wn[1], wn[0]))
    return {
        "words": total,
        "unique": len(counts),
        "sentences": sentence_count(text),
        "average_length": round(letters / total, 1) if total else 0,
        "longest": sorted(counts, key=lambda w: (-len(w), w))[:3],
        "common": [[w, n] for w, n in common[:top]],
    }


def build_parser():
    parser = argparse.ArgumentParser(prog="textstats", description="Word statistics for a text file")
    parser.add_argument("path")
    parser.add_argument("--top", type=int, default=5)
    parser.add_argument("--json", action="store_true")
    parser.add_argument("--stopwords", action="store_true")
    return parser


def main(argv):
    args = build_parser().parse_args(argv)
    try:
        with open(args.path, encoding="utf-8") as f:
            text = f.read()
    except FileNotFoundError:
        print(f"textstats: no such file: {args.path}", file=sys.stderr)
        return 1
    result = analyse(text, top=args.top, stopwords=STOPWORDS if args.stopwords else ())
    if args.json:
        print(json.dumps(result))
        return 0
    print(f"{result['words']} words, {result['unique']} unique, {result['sentences']} sentences")
    print(f"Average word length: {result['average_length']}")
    print("Longest: " + ", ".join(result["longest"]))
    print("Most common:")
    for word, count in result["common"]:
        print(f"  {word:<12}{count:>4}")
    return 0
