## Integer types

C's integer types have **minimum** sizes, not exact ones: an `int` is at least 16 bits, in practice 32 on today's desktops. `sizeof` tells you the size in bytes on *this* machine:

```c
#include <stdio.h>
#include <limits.h>
#include <stdint.h>

int main(void) {
    printf("char %zu, short %zu, int %zu, long %zu, long long %zu bytes\n",
           sizeof(char), sizeof(short), sizeof(int), sizeof(long), sizeof(long long));
    printf("int range: %d to %d\n", INT_MIN, INT_MAX);
    printf("unsigned int max: %u\n", UINT_MAX);

    int32_t exact = 2000000000;      // exactly 32 bits (stdint.h)
    uint8_t byte = 255;              // exactly 8 bits, unsigned
    int64_t big = 9000000000;        // 64 bits for big values
    printf("%d %d %lld\n", (int)exact, byte, (long long)big);
    return 0;
}
```

- `long` is 32 bits on Windows but 64 bits on Linux and macOS: one reason portable code uses the **fixed-width types** of `<stdint.h>`: `int8_t` … `int64_t`, `uint8_t` … `uint64_t`, and `size_t` for sizes and indexes.
- **`unsigned`** types hold only non-negative values and **wrap around** (modulo 2ⁿ): `0u - 1` is `UINT_MAX`. That's defined behaviour.
- **Signed overflow is undefined behaviour**: `INT_MAX + 1` is not guaranteed to wrap. Check before you add if values can be large.
- `char` is a small integer (one byte) used for characters: `'A'` is the number 65.

```c
#include <stdio.h>
#include <limits.h>

int main(void) {
    unsigned int u = 0;
    u = u - 1;                         // wraps: defined for unsigned
    printf("0u - 1 = %u\n", u);

    char letter = 'A';
    printf("%c is %d; next is %c\n", letter, letter, letter + 1);

    int a = INT_MAX;
    if (a > INT_MAX - 1) {            // check BEFORE adding, never after
        printf("adding 1 to %d would overflow\n", a);
    }
    return 0;
}
```

## Floating point

`float` (about 7 significant digits) and `double` (about 15–16) store binary approximations; **use `double`** by default. Most decimals can't be represented exactly:

```c
#include <stdio.h>
#include <math.h>

int main(void) {
    double x = 0.1 + 0.2;
    printf("%.17f\n", x);
    printf("equal to 0.3? %d\n", x == 0.3);
    printf("close to 0.3? %d\n", fabs(x - 0.3) < 1e-9);
    float f = 16777217.0f;             // float can't hold every integer above 2^24
    printf("%.1f\n", f);
    printf("%f %f\n", sqrt(2.0), pow(2.0, 10));
    return 0;
}
```

Compare floating-point numbers with a tolerance, never with `==`. Store money as integer cents.

## bool

Since C99, `<stdbool.h>` provides `bool`, `true` and `false` (C23 makes them keywords). Underneath, conditions are integers: **0 is false, anything else is true**.

```c
#include <stdio.h>
#include <stdbool.h>

int main(void) {
    bool ready = true;
    int count = 5;
    if (count) {                       // non-zero: true
        printf("count is non-zero\n");
    }
    printf("ready=%d, !ready=%d, (3 > 2)=%d\n", ready, !ready, 3 > 2);
    return 0;
}
```

=== lesson c-operators | Operators, expressions and conversions | 65
## Arithmetic

```c
#include <stdio.h>

int main(void) {
    int a = 17, b = 5;
    printf("%d %d %d %d %d\n", a + b, a - b, a * b, a / b, a % b);
    printf("%d %d\n", -17 / 5, -17 % 5);     // division truncates toward zero
    printf("%.2f\n", (double)a / b);          // convert before dividing
    printf("%.2f\n", (double)(a / b));        // too late: 3.00
    return 0;
}
```

Integer division by zero is **undefined behaviour** (usually a crash); floating-point division by zero gives infinity.

## Increment, assignment and side effects

```c
#include <stdio.h>

int main(void) {
    int i = 5;
    int a = i++;   // a = 5, then i becomes 6 (postfix: use, then increment)
    int b = ++i;   // i becomes 7, then b = 7 (prefix: increment, then use)
    printf("a=%d b=%d i=%d\n", a, b, i);

    int total = 10;
    total += 5;    // also -=, *=, /=, %=, <<=, &=, |=...
    total *= 2;
    printf("total=%d\n", total);
    return 0;
}
```

Never modify the same variable twice in one expression (`i = i++ + ++i;`): the result is undefined. Keep side effects in separate statements.

## Comparison, logic and the conditional operator

Comparisons (`== != < <= > >=`) give `int` 1 or 0. `&&` and `||` **short-circuit**: the right side is evaluated only when needed, which makes guards like `p != NULL && p->count > 0` safe. `cond ? a : b` picks a value:

```c
#include <stdio.h>

int main(void) {
    int age = 17;
    int has_parent = 1;
    int can_enter = age >= 18 || has_parent;
    printf("can enter: %d\n", can_enter);
    printf("%s\n", age >= 18 ? "adult" : "minor");
    int x = 0;
    if (x != 0 && 10 / x > 1) { // safe: 10 / x never runs when x is 0
        printf("big\n");
    }
    return 0;
}
```

The classic C bug: `if (x = 5)` **assigns** 5 (true), instead of comparing. Compilers warn about it (with `-Wall`).

## Bitwise operators

C is the language of bits: flags, hardware registers, network protocols, compression.

```c
#include <stdio.h>
#include <stdint.h>

#define FLAG_READ  (1u << 0)
#define FLAG_WRITE (1u << 1)
#define FLAG_EXEC  (1u << 2)

int main(void) {
    unsigned int perms = FLAG_READ | FLAG_WRITE;    // set bits
    printf("can write: %d\n", (perms & FLAG_WRITE) != 0);
    perms &= ~FLAG_WRITE;                           // clear a bit
    perms ^= FLAG_EXEC;                             // toggle a bit
    printf("perms = %u (0x%x)\n", perms, perms);

    uint32_t color = 0x11AAFF;
    unsigned red = (color >> 16) & 0xFF, green = (color >> 8) & 0xFF, blue = color & 0xFF;
    printf("r=%u g=%u b=%u\n", red, green, blue);
    return 0;
}
```

Use **unsigned** types for bit manipulation: shifting negative numbers or shifting into the sign bit is undefined or implementation-defined.

## Implicit conversions and casts

In mixed expressions C converts operands to a common type: `int` + `double` → `double`. Small types (`char`, `short`) are **promoted** to `int` first. When signed meets unsigned of the same size, the signed value is converted to unsigned, which causes the famous `-1 > 0u` surprise. A **cast** `(type)value` converts explicitly:

```c
#include <stdio.h>

int main(void) {
    double price = 9.99;
    int euros = (int)price;           // truncates: 9
    int cents = (int)(price * 100 + 0.5);
    printf("%d euros, %d cents in total\n", euros, cents);

    unsigned int u = 1;
    int s = -1;
    printf("-1 < 1u? %d\n", s < (int)u); // cast so both are signed
    char c = 'a';
    printf("'a' + 1 = %d (%c)\n", c + 1, c + 1);
    return 0;
}
```

Precedence has 15 levels; the surprises are bitwise operators being **lower** than comparisons (`flags & MASK == 0` means `flags & (MASK == 0)`). When in doubt, add parentheses.

=== lesson c-io | Input and output | 70
## printf in depth

You've used `printf`; two more members of the family are essential: **`fprintf(stderr, ...)`** writes error messages to the standard error stream (so they don't mix with normal output), and **`snprintf(buf, size, ...)`** formats into a string, never writing more than `size` bytes:

```c
#include <stdio.h>

int main(void) {
    char label[16];
    int written = snprintf(label, sizeof label, "item-%03d", 7);
    printf("%s (%d chars)\n", label, written);

    char small[6];
    snprintf(small, sizeof small, "%s", "truncated text"); // safely cut to 5 chars + '\0'
    printf("%s\n", small);

    fprintf(stderr, "this goes to the error stream\n");
    return 0;
}
```

(In a terminal you'd see the `stderr` line too; DevQuest shows only standard output for a program that succeeds, which is exactly why errors belong on `stderr`: they don't pollute the real output.)

## Reading input: the safe way

`scanf` is the textbook function, but it's full of traps: it leaves the line break in the input, stops at spaces, and `scanf("%s", buf)` can overflow `buf`. The robust pattern is: **read a whole line with `fgets`, then parse it**:

```c
// stdin: Ada Lovelace | 36
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main(void) {
    char name[64];
    char line[64];

    printf("Name: ");
    if (fgets(name, sizeof name, stdin) == NULL) return 1;
    name[strcspn(name, "\n")] = '\0';          // remove the line break fgets keeps

    printf("Age: ");
    if (fgets(line, sizeof line, stdin) == NULL) return 1;
    char *end;
    long age = strtol(line, &end, 10);         // parse, and know where it stopped
    if (end == line) {
        printf("\nthat's not a number\n");
        return 1;
    }
    printf("\nHello %s, next year you'll be %ld.\n", name, age + 1);
    return 0;
}
```

- **`fgets(buf, size, stdin)`** reads at most `size - 1` characters, stops after a line break (which it keeps) and adds the terminating `'\0'`. It returns `NULL` at the end of the input.
- **`strtol(text, &end, 10)`** converts text to a `long` and tells you where it stopped (`end`), so you can detect garbage; `strtod` does the same for doubles. (`atoi` exists but can't report errors.)
- **`sscanf(line, "%d %d", &a, &b)`** parses a line you've already read, and returns how many values it converted: always check it.

## scanf, if you use it

```c
// stdin: 12 30
#include <stdio.h>

int main(void) {
    int hours, minutes;
    printf("Time (h m): ");
    if (scanf("%d %d", &hours, &minutes) != 2) {   // returns the number of items read
        printf("\nexpected two numbers\n");
        return 1;
    }
    printf("\n%d minutes in total\n", hours * 60 + minutes);
    return 0;
}
```

Note the **`&`**: `scanf` needs the address of each variable to store into it (pointers, Module 4). Forgetting it is a classic crash. Never use `scanf("%s", ...)` without a width (`%63s`), and never use the removed function `gets`. (Microsoft's compiler goes further and calls `scanf`, `strcpy` and `fopen` "deprecated", suggesting `scanf_s` and friends; those aren't portable C, so DevQuest turns that particular warning off with `_CRT_SECURE_NO_WARNINGS`.)

## Reading until the end

```c
// stdin: 4 | 8 | x | 15
#include <stdio.h>

int main(void) {
    char line[64];
    int sum = 0, count = 0;
    while (fgets(line, sizeof line, stdin) != NULL) {
        int n;
        if (sscanf(line, "%d", &n) == 1) {
            sum += n;
            count++;
        }
    }
    printf("%d numbers, sum %d\n", count, sum);
    return 0;
}
```

::solution
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

int main(void) {
    char line[128];
    long long balance = 0, in = 0, out = 0;
    while (fgets(line, sizeof line, stdin) != NULL) {
        line[strcspn(line, "\n")] = '\0';
        char *rest;
        double value = strtod(line, &rest);
        char description[32];
        if (rest == line || sscanf(rest, "%31s", description) != 1) {
            printf("skipped: %s\n", line);
            continue;
        }
        long long cents = llround(value * 100);
        balance += cents;
        if (cents >= 0) {
            in += cents;
        } else {
            out -= cents;
        }
        printf("%+-9.2f %-15s balance %8.2f\n", cents / 100.0, description, balance / 100.0);
    }
    printf("in: %.2f, out: %.2f, final: %.2f\n", in / 100.0, out / 100.0, balance / 100.0);
    return 0;
}
