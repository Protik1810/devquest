## FILE streams

The standard library reads and writes files through **`FILE *`** streams. `fopen` opens one and returns `NULL` on failure; `fclose` flushes and closes it:

```c
#include <stdio.h>
#include <string.h>
#include <errno.h>

int main(void) {
    FILE *out = fopen("notes.txt", "w");          // "w": create or truncate for writing
    if (out == NULL) {
        perror("notes.txt");                      // prints the reason, from errno
        return 1;
    }
    fprintf(out, "buy milk\n");
    fputs("write report\n", out);
    fclose(out);

    FILE *app = fopen("notes.txt", "a");          // "a": append
    if (app == NULL) return 1;
    fprintf(app, "call %s\n", "Ada");
    fclose(app);

    FILE *in = fopen("notes.txt", "r");           // "r": read
    if (in == NULL) return 1;
    char line[128];
    int n = 0;
    while (fgets(line, sizeof line, in) != NULL) {
        line[strcspn(line, "\n")] = '\0';
        printf("%d: %s\n", ++n, line);
    }
    fclose(in);

    FILE *missing = fopen("no-such-file.txt", "r");
    if (missing == NULL) {
        printf("cannot open: %s\n", strerror(errno));
    }
    remove("notes.txt");                          // delete the file
    return 0;
}
```

| Mode | Meaning |
|---|---|
| `"r"` | read; the file must exist |
| `"w"` | write; creates or **empties** the file |
| `"a"` | append at the end; creates if needed |
| `"r+"`, `"w+"`, `"a+"` | read and write |
| add `"b"` | binary mode (`"rb"`, `"wb"`): no line-ending translation on Windows |

When a library function fails it sets **`errno`**; `perror(prefix)` and `strerror(errno)` turn it into a message. Every open file must be closed exactly once, like every `malloc` needs a `free`.

## Binary files

Text files are for humans; **binary** files store the bytes of your data directly. `fwrite` and `fread` copy whole arrays of structs, and `fseek`/`ftell` move around the file:

```c
#include <stdio.h>

typedef struct {
    int id;
    double score;
} Record;

int main(void) {
    Record records[] = {{1, 9.5}, {2, 7.25}, {3, 8.0}};
    FILE *f = fopen("scores.bin", "wb");
    if (f == NULL) return 1;
    size_t written = fwrite(records, sizeof records[0], 3, f);
    fclose(f);
    printf("wrote %zu records\n", written);

    f = fopen("scores.bin", "rb");
    if (f == NULL) return 1;
    fseek(f, 0, SEEK_END);
    long size = ftell(f);                         // file size in bytes
    printf("file is %ld bytes (%zu per record)\n", size, sizeof(Record));

    fseek(f, (long)sizeof(Record), SEEK_SET);     // jump straight to the second record
    Record r;
    if (fread(&r, sizeof r, 1, f) == 1) {
        printf("record %d: %.2f\n", r.id, r.score);
    }
    fclose(f);
    remove("scores.bin");
    return 0;
}
```

Raw struct dumps are fast but **not portable**: padding, integer sizes and byte order (endianness) differ between machines. File formats meant to be shared define every byte explicitly, or use text (CSV, JSON).

## Standard streams

`stdin`, `stdout` and `stderr` are `FILE *` too: `fprintf(stderr, ...)`, `fgets(buf, n, stdin)`. Output to a file or a pipe is **buffered**: data is written in blocks, so call `fflush(stdout)` when output must appear right now (for example, before a crash you're investigating).

## Before the compiler: the preprocessor

Lines starting with `#` are handled by the **preprocessor**, a text-substitution pass that runs before compilation:

```c
#include <stdio.h>

#define PI 3.14159265358979
#define SQUARE(x) ((x) * (x))              // a function-like macro: parenthesise everything
#define BAD_SQUARE(x) x * x
#define MAX(a, b) ((a) > (b) ? (a) : (b))
#define ARRAY_LEN(a) (sizeof (a) / sizeof (a)[0])
#define DEBUG 1

int main(void) {
    printf("area %.2f\n", PI * SQUARE(2.0));
    printf("SQUARE(1 + 2) = %d, BAD_SQUARE(1 + 2) = %d\n", SQUARE(1 + 2), BAD_SQUARE(1 + 2));
    int i = 3;
    int m = MAX(i, 2);
    printf("max %d\n", m);
    int data[] = {4, 8, 15, 16, 23, 42};
    printf("%zu elements\n", ARRAY_LEN(data));
#if DEBUG
    printf("[debug] %s line %d, compiled on %s\n", __FILE__, __LINE__, __DATE__);
#endif
    return 0;
}
```

- `BAD_SQUARE(1 + 2)` expands to `1 + 2 * 1 + 2`, which is 5: macros substitute **text**, not values. Parenthesise every parameter and the whole body.
- A macro argument may be evaluated **twice**: `MAX(i++, 2)` increments `i` twice. Prefer `static inline` functions and `const` or `enum` constants when you can; use macros for what functions can't do (`ARRAY_LEN`, conditional compilation, `__FILE__` and `__LINE__`).
- `#if`, `#ifdef`, `#ifndef`, `#elif`, `#else`, `#endif` include or exclude code, for debugging or for different platforms (`#ifdef _WIN32`). Compilers let you define macros on the command line: `gcc -DDEBUG=1`.

## Splitting a program into files

Real C programs are split into **modules**: a header (`.h`) declares the public interface, and a source file (`.c`) implements it.

```
// stack.h — the interface
#ifndef STACK_H              // include guard: the header's content is seen only once
#define STACK_H

#include <stddef.h>

typedef struct Stack Stack;  // opaque type: users can't see the fields

Stack *stack_new(void);
void stack_free(Stack *s);
int stack_push(Stack *s, int value);
int stack_pop(Stack *s, int *out);
size_t stack_size(const Stack *s);

#endif
```

```
// stack.c — the implementation
#include <stdlib.h>
#include "stack.h"           // quotes: search the project's folders first

struct Stack {               // the full definition is private to this file
    int *data;
    size_t len, cap;
};

static int grow(Stack *s) {  // static: not visible outside stack.c
    size_t cap = s->cap ? s->cap * 2 : 8;
    int *data = realloc(s->data, cap * sizeof *data);
    if (data == NULL) return -1;
    s->data = data;
    s->cap = cap;
    return 0;
}

Stack *stack_new(void) { return calloc(1, sizeof(Stack)); }
/* ... the other functions ... */
```

```
// main.c
#include <stdio.h>
#include "stack.h"

int main(void) {
    Stack *s = stack_new();
    stack_push(s, 42);
    printf("%zu\n", stack_size(s));
    stack_free(s);
}
```

Build it with `gcc -Wall -Wextra -o app main.c stack.c`. The **opaque type** is C's encapsulation: code outside `stack.c` can only use the functions, so the representation can change freely.

- `extern int counter;` in a header **declares** a global defined (once) in one `.c` file. Defining a non-static global in a header gives "multiple definition" linker errors.
- `static` functions and globals are private to their file.

## Build tools

Recompiling everything by hand gets slow. **make** rebuilds only what changed, from rules in a `Makefile`:

```
CC = gcc
CFLAGS = -std=c17 -Wall -Wextra -g

app: main.o stack.o
	$(CC) $(CFLAGS) -o app main.o stack.o

%.o: %.c stack.h
	$(CC) $(CFLAGS) -c $<

clean:
	rm -f app *.o
```

**CMake** generates build files for any platform and IDE (`add_executable(app main.c stack.c)`), and is the usual choice for larger and cross-platform projects.

## Sorting and searching

`qsort` sorts any array, and `bsearch` searches a sorted one, given a **comparison function** that receives pointers to two elements as `const void *`:

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char name[16];
    int score;
} Player;

static int by_score_desc(const void *a, const void *b) {
    const Player *p = a, *q = b;
    if (p->score != q->score) return (q->score > p->score) - (q->score < p->score);
    return strcmp(p->name, q->name);
}

static int cmp_int(const void *a, const void *b) {
    int x = *(const int *)a, y = *(const int *)b;
    return (x > y) - (x < y);
}

int main(void) {
    Player players[] = {{"cy", 70}, {"ada", 95}, {"bob", 70}, {"dee", 88}};
    size_t n = sizeof players / sizeof players[0];
    qsort(players, n, sizeof players[0], by_score_desc);
    for (size_t i = 0; i < n; i++) printf("%s %d\n", players[i].name, players[i].score);

    int primes[] = {2, 3, 5, 7, 11, 13, 17, 19};
    int key = 13;
    int *found = bsearch(&key, primes, 8, sizeof primes[0], cmp_int);
    printf("13 %s at index %td\n", found ? "found" : "missing", found ? found - primes : -1);
    return 0;
}
```

`void *` is C's generic pointer: any object pointer converts to and from it without a cast. That's how `qsort`, `malloc` and `memcpy` work with any type.

```c
#include <stdio.h>
#include <time.h>
#include <math.h>
#include <assert.h>

int main(void) {
    struct tm date = {.tm_year = 2026 - 1900, .tm_mon = 2, .tm_mday = 15, .tm_hour = 9};
    time_t t = mktime(&date);                      // normalises the fields and fills the weekday
    char buf[64];
    strftime(buf, sizeof buf, "%A %d %B %Y, %H:%M", &date);
    printf("%s\n", buf);

    struct tm later = date;
    later.tm_mday += 30;                           // mktime handles month overflow
    mktime(&later);
    strftime(buf, sizeof buf, "%Y-%m-%d", &later);
    printf("30 days later: %s (%.0f days apart)\n", buf, difftime(mktime(&later), t) / 86400);

    printf("%.3f %.1f %.1f %.1f\n", hypot(3, 4), floor(-2.5), round(2.5), fmod(10, 3));
    assert(sqrt(16.0) == 4.0);
    printf("assertions passed\n");
    return 0;
}
```

## Random numbers

`rand()` returns pseudo-random numbers from 0 to `RAND_MAX`, seeded with `srand(seed)` (often `srand((unsigned)time(NULL))`). It's fine for games and simulations, but its quality and sequence differ between C libraries, and it must **never** be used for passwords or keys (use the operating system's cryptographic generator). For reproducible tests, write a small generator yourself, like this one (a *xorshift*):

```c
#include <stdio.h>
#include <stdint.h>

static uint32_t state = 2463534242u;

uint32_t xorshift32(void) {
    state ^= state << 13;
    state ^= state >> 17;
    state ^= state << 5;
    return state;
}

int main(void) {
    for (int i = 0; i < 5; i++) {
        printf("dice: %u\n", xorshift32() % 6 + 1);
    }
    return 0;
}
```

## Ending a program

`return` from `main` or **`exit(status)`** anywhere ends the program normally: buffers are flushed and functions registered with `atexit` run. `abort()` ends it immediately (and `assert` uses it).

## C has evolved

Code written in 1989 style still compiles, but modern C is safer and clearer:

| Standard | Notable additions |
|---|---|
| **C99** | `//` comments, declarations anywhere (`for (int i = 0; ...)`), `bool`, `<stdint.h>`, `long long`, designated initialisers, compound literals, `inline`, `snprintf`, variable-length arrays |
| **C11** | `_Static_assert`, anonymous structs and unions, `_Generic`, `<threads.h>` and atomics, `aligned_alloc`, removal of `gets` |
| **C17** | bug fixes |
| **C23** | `bool`, `true`, `false`, `nullptr` and `static_assert` as keywords, `constexpr`, `typeof`, `[[nodiscard]]` and other attributes, binary literals `0b1010`, digit separators `1'000'000`, `#embed` |

```c
#include <stdio.h>
#include <stdint.h>
#include <stdbool.h>

typedef struct {
    int x, y;
} Point;

_Static_assert(sizeof(int32_t) == 4, "int32_t must be 4 bytes"); // checked at compile time

static int manhattan(Point a, Point b) {
    int dx = a.x - b.x, dy = a.y - b.y;
    return (dx < 0 ? -dx : dx) + (dy < 0 ? -dy : dy);
}

#define type_name(x) _Generic((x), int: "int", double: "double", const char *: "string", default: "other")

int main(void) {
    Point p = {.x = 3, .y = 4};                                 // designated initialiser
    int d = manhattan(p, (Point){.x = 0, .y = 0});              // compound literal: an unnamed object
    bool far = d > 5;
    printf("distance %d, far %d\n", d, far);
    printf("%s %s %s\n", type_name(42), type_name(2.5), type_name((const char *)"hi"));
    int squares[5] = {[1] = 1, [4] = 16};                       // designated array elements
    printf("%d %d %d\n", squares[0], squares[1], squares[4]);
    return 0;
}
```

## Undefined behaviour: the list to remember

The compiler assumes your program **never** does any of these, and optimises accordingly. Code with undefined behaviour can work in a debug build and break in an optimised one:

- reading or writing out of bounds (arrays, strings, heap blocks);
- using memory after `free`, freeing twice, freeing what `malloc` didn't return;
- dereferencing `NULL` or an uninitialised pointer;
- reading an uninitialised variable;
- signed integer overflow, shifting by a negative amount or by the type's width or more;
- integer division by zero (and `INT_MIN / -1`);
- modifying a string literal or a `const` object;
- modifying a variable twice in one expression (`i = i++`);
- a `printf`/`scanf` specifier that doesn't match the argument's type;
- violating *strict aliasing* (reading a `float` through an `int *`): use `memcpy` instead.

## Writing safe C

Most security vulnerabilities in C code come from a few patterns. The defences are habits:

1. **Bounds**: always carry sizes with buffers; check indexes; use `snprintf` instead of `sprintf`/`strcpy`/`strcat`; never `gets`.
2. **Integer overflow**: check before arithmetic that can overflow, especially sizes passed to `malloc` (`n * size`).
3. **Never use input as a format string**: `printf(user_text)` lets an attacker read memory with `%x` and even write it with `%n`. Write `printf("%s", user_text)`.
4. **Memory**: one owner per allocation, `NULL` after `free`, the `goto cleanup` pattern.
5. **Tools**: `-Wall -Wextra -Werror`, AddressSanitizer and UndefinedBehaviorSanitizer while testing, static analysers (clang-tidy, cppcheck), fuzzing (libFuzzer, AFL++).
6. **const** everything that doesn't change, and give each function one clear job.

```c
#include <stdio.h>
#include <stdint.h>
#include <stdlib.h>

// returns 0 and stores a * b in *out, or -1 if the product would overflow size_t
int mul_size(size_t a, size_t b, size_t *out) {
    if (a != 0 && b > SIZE_MAX / a) return -1;
    *out = a * b;
    return 0;
}

void *alloc_array(size_t count, size_t elem_size) {
    size_t bytes;
    if (mul_size(count, elem_size, &bytes) != 0) return NULL;   // refuse instead of wrapping
    return malloc(bytes);
}

int main(void) {
    void *ok = alloc_array(1000, sizeof(int));
    void *huge = alloc_array(SIZE_MAX / 2, 4);
    printf("ok allocated: %d, huge refused: %d\n", ok != NULL, huge == NULL);
    free(ok);

    const char *user_text = "100% safe";
    printf("%s\n", user_text);       // never printf(user_text)
    return 0;
}
```

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define BUF_MAX 1024

typedef struct {
    char *data;
    size_t len, cap;
} Buf;

int buf_append(Buf *b, const char *text, size_t text_len) {
    if (text_len > SIZE_MAX - b->len - 1) return -1;   // len + text_len + 1 would overflow
    size_t needed = b->len + text_len + 1;
    if (needed > BUF_MAX) return -1;
    if (needed > b->cap) {
        size_t cap = b->cap ? b->cap : 16;
        while (cap < needed) cap *= 2;
        char *data = realloc(b->data, cap);
        if (data == NULL) return -1;
        b->data = data;
        b->cap = cap;
    }
    memcpy(b->data + b->len, text, text_len);
    b->len += text_len;
    b->data[b->len] = '\0';
    return 0;
}

int buf_append_int(Buf *b, long value) {
    char tmp[24];
    int n = snprintf(tmp, sizeof tmp, "%ld", value);
    if (n < 0 || (size_t)n >= sizeof tmp) return -1;
    return buf_append(b, tmp, (size_t)n);
}

void buf_free(Buf *b) {
    free(b->data);
    *b = (Buf){0};
}

int main(void) {
    Buf b = {0};
    buf_append(&b, "hello, world", 12);
    printf("%s (%zu/%zu)\n", b.data, b.len, b.cap);
    buf_append(&b, " count: ", 8);
    buf_append_int(&b, 42);
    buf_append(&b, " ", 1);
    buf_append_int(&b, -7);
    printf("%s (%zu/%zu)\n", b.data + 13, b.len, b.cap);
    char big[2000];
    memset(big, 'x', sizeof big);
    printf("too big: %d, still %zu bytes\n", buf_append(&b, big, sizeof big), b.len);
    printf("overflow: %d\n", buf_append(&b, "x", SIZE_MAX));
    printf("data still ok: %s\n", b.data);
    buf_free(&b);
    return 0;
}

#include <stdio.h>
#include <string.h>

int analyse(const char *path, FILE *out) {
    FILE *in = fopen(path, "r");
    if (in == NULL) {
        fprintf(out, "error: cannot open %s\n", path);
        return -1;
    }
    int total = 0, malformed = 0, info = 0, warn = 0, error = 0;
    char first[48] = "", last[48] = "";
    char comps[8][32];
    int comp_errors[8];
    int ncomp = 0;
    char line[256];
    while (fgets(line, sizeof line, in) != NULL) {
        total++;
        char date[11], time[9], level[8], component[32], message[128];
        if (sscanf(line, "%10s %8s %7s %31[^:]: %127[^\n]", date, time, level, component, message) != 5) {
            malformed++;
            continue;
        }
        if (strcmp(level, "INFO") == 0) {
            info++;
        } else if (strcmp(level, "WARN") == 0) {
            warn++;
        } else if (strcmp(level, "ERROR") == 0) {
            error++;
            snprintf(last, sizeof last, "%s %s", time, component);
            if (first[0] == '\0') snprintf(first, sizeof first, "%s", last);
            int i = 0;
            while (i < ncomp && strcmp(comps[i], component) != 0) i++;
            if (i == ncomp && ncomp < 8) {
                snprintf(comps[ncomp], sizeof comps[0], "%s", component);
                comp_errors[ncomp++] = 0;
            }
            if (i < ncomp) comp_errors[i]++;
        } else {
            malformed++;
        }
    }
    fclose(in);
    fprintf(out, "lines: %d (malformed: %d)\n", total, malformed);
    fprintf(out, "INFO %d, WARN %d, ERROR %d\n", info, warn, error);
    if (error > 0) {
        fprintf(out, "first error: %s\n", first);
        fprintf(out, "last error: %s\n", last);
    }
    fprintf(out, "errors by component:");
    for (int i = 0; i < ncomp; i++) fprintf(out, " %s=%d", comps[i], comp_errors[i]);
    fprintf(out, "\n");
    return malformed;
}
