## Why write them yourself?

`qsort` and `bsearch` cover everyday needs, but knowing how the classic algorithms work, and how fast they are, is part of being a professional programmer: it guides your choice of data structures, and it's a staple of interviews. C is the ideal language to see them clearly, because nothing is hidden.

## Measuring cost: big O

**Big O** describes how the work grows with the input size n:

| Complexity | Example | 1,000 items | 1,000,000 items |
|---|---|---|---|
| O(1) | array indexing, hash lookup | 1 | 1 |
| O(log n) | binary search | ~10 | ~20 |
| O(n) | linear search, summing | 1,000 | 1,000,000 |
| O(n log n) | merge sort, quicksort | ~10,000 | ~20,000,000 |
| O(n²) | insertion sort, comparing all pairs | 1,000,000 | 10¹² (hours) |

## Binary search

On a **sorted** array, compare with the middle element and discard half of the range each time:

```c
#include <stdio.h>

// returns the index of target, or -1
int binary_search(const int *a, int n, int target) {
    int lo = 0, hi = n - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;   // not (lo + hi) / 2, which can overflow
        if (a[mid] == target) return mid;
        if (a[mid] < target) lo = mid + 1;
        else hi = mid - 1;
    }
    return -1;
}

int main(void) {
    int a[] = {2, 5, 8, 12, 16, 23, 38, 56, 72, 91};
    printf("%d %d %d\n", binary_search(a, 10, 23), binary_search(a, 10, 2), binary_search(a, 10, 7));
    return 0;
}
```

## Merge sort: divide and conquer

Split the array in two, sort each half recursively, then **merge** the two sorted halves. Always O(n log n), and **stable** (equal elements keep their order), but it needs a temporary buffer:

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

static void merge_sort_rec(int *a, int *tmp, size_t lo, size_t hi) { // sorts a[lo, hi)
    if (hi - lo < 2) return;
    size_t mid = lo + (hi - lo) / 2;
    merge_sort_rec(a, tmp, lo, mid);
    merge_sort_rec(a, tmp, mid, hi);
    size_t i = lo, j = mid, k = lo;
    while (i < mid && j < hi) tmp[k++] = a[i] <= a[j] ? a[i++] : a[j++];
    while (i < mid) tmp[k++] = a[i++];
    while (j < hi) tmp[k++] = a[j++];
    memcpy(a + lo, tmp + lo, (hi - lo) * sizeof *a);
}

int merge_sort(int *a, size_t n) {
    int *tmp = malloc(n * sizeof *tmp);
    if (tmp == NULL) return -1;
    merge_sort_rec(a, tmp, 0, n);
    free(tmp);
    return 0;
}

int main(void) {
    int a[] = {38, 27, 43, 3, 9, 82, 10, 3};
    merge_sort(a, 8);
    for (int i = 0; i < 8; i++) printf("%d ", a[i]);
    printf("\n");
    return 0;
}
```

## Quicksort

Pick a **pivot**, move smaller elements before it and larger ones after (**partition**), then sort both sides recursively. On average O(n log n) and very fast in practice, in place, but O(n²) on unlucky inputs (for example an already sorted array with the first element as pivot), which is why real implementations pick pivots carefully. `qsort` is usually a tuned quicksort or introsort.

```c
#include <stdio.h>

static void swap(int *x, int *y) {
    int t = *x;
    *x = *y;
    *y = t;
}

// Lomuto partition with the middle element as pivot
static void quicksort(int *a, int lo, int hi) {   // sorts a[lo..hi]
    if (lo >= hi) return;
    swap(&a[lo + (hi - lo) / 2], &a[hi]);
    int pivot = a[hi], store = lo;
    for (int i = lo; i < hi; i++) {
        if (a[i] < pivot) swap(&a[i], &a[store++]);
    }
    swap(&a[store], &a[hi]);
    quicksort(a, lo, store - 1);
    quicksort(a, store + 1, hi);
}

int main(void) {
    int a[] = {9, -3, 5, 2, 6, 8, -6, 1, 3};
    quicksort(a, 0, 8);
    for (int i = 0; i < 9; i++) printf("%d ", a[i]);
    printf("\n");
    return 0;
}
```

#include <stdio.h>
#include <string.h>

static long comparisons = 0;

void insertion_sort(int *a, int n) {
    for (int i = 1; i < n; i++) {
        int key = a[i];
        int j = i;
        while (j > 0) {
            comparisons++;
            if (a[j - 1] <= key) break;
            a[j] = a[j - 1];
            j--;
        }
        a[j] = key;
    }
}

void merge_sort(int *a, int *tmp, int lo, int hi) {
    if (hi - lo < 2) return;
    int mid = lo + (hi - lo) / 2;
    merge_sort(a, tmp, lo, mid);
    merge_sort(a, tmp, mid, hi);
    int i = lo, j = mid, k = lo;
    while (i < mid && j < hi) {
        comparisons++;
        tmp[k++] = a[i] <= a[j] ? a[i++] : a[j++];
    }
    while (i < mid) tmp[k++] = a[i++];
    while (j < hi) tmp[k++] = a[j++];
    memcpy(a + lo, tmp + lo, (size_t)(hi - lo) * sizeof *a);
}

int linear_search(const int *a, int n, int target) {
    for (int i = 0; i < n; i++) {
        comparisons++;
        if (a[i] == target) return i;
    }
    return -1;
}

int binary_search(const int *a, int n, int target) {
    int lo = 0, hi = n - 1, found = -1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;
        comparisons++;
        if (a[mid] == target) {
            found = mid;
            lo = mid + 1; // keep looking right, to find the last equal element
        } else if (a[mid] < target) {
            lo = mid + 1;
        } else {
            hi = mid - 1;
        }
    }
    return found;
}

int main(void) {

    for (int i = 0; i < 64; i++) {

## How integers are stored

Integers are stored in **two's complement**: the highest bit has a negative weight, so for 8 bits `11111111` is -1 and `10000000` is -128. Larger integers take several bytes, and processors disagree on their **order**:

- **little-endian** (x86, ARM in practice): the least significant byte first;
- **big-endian** ("network byte order"): the most significant byte first, as in most file formats and network protocols.

```c
#include <stdio.h>
#include <stdint.h>
#include <string.h>

int main(void) {
    uint32_t value = 0x11223344;
    unsigned char bytes[4];
    memcpy(bytes, &value, sizeof value);         // look at the bytes as stored in memory
    printf("in memory: %02x %02x %02x %02x -> %s-endian\n", bytes[0], bytes[1], bytes[2], bytes[3],
           bytes[0] == 0x44 ? "little" : "big");

    int8_t minus_one = -1;
    printf("-1 as a byte: 0x%02x, -128 as a byte: 0x%02x\n", (uint8_t)minus_one, (uint8_t)INT8_MIN);
    return 0;
}
```

## Portable serialisation

Writing a struct with `fwrite` copies its in-memory layout (padding, endianness), which differs between machines. Portable formats write each field **byte by byte in a defined order**, with shifts, and read it back the same way:

```c
#include <stdio.h>
#include <stdint.h>

void put_u32_be(unsigned char *out, uint32_t v) {   // big-endian, whatever the CPU
    out[0] = (unsigned char)(v >> 24);
    out[1] = (unsigned char)(v >> 16);
    out[2] = (unsigned char)(v >> 8);
    out[3] = (unsigned char)v;
}

uint32_t get_u32_be(const unsigned char *in) {
    return (uint32_t)in[0] << 24 | (uint32_t)in[1] << 16 | (uint32_t)in[2] << 8 | in[3];
}

int main(void) {
    unsigned char buf[4];
    put_u32_be(buf, 3000000000u);
    printf("%02X %02X %02X %02X -> %u\n", buf[0], buf[1], buf[2], buf[3], get_u32_be(buf));
    return 0;
}
```

The `(uint32_t)` casts matter: `in[0] << 24` would shift an `int`, and shifting into its sign bit is undefined.

## Bit tricks worth knowing

```c
#include <stdio.h>
#include <stdint.h>

int popcount(uint32_t x) {            // number of 1 bits
    int n = 0;
    while (x) {
        x &= x - 1;                   // clears the lowest set bit
        n++;
    }
    return n;
}

int is_power_of_two(uint32_t x) { return x != 0 && (x & (x - 1)) == 0; }

uint32_t round_up_pow2(uint32_t x) {  // smallest power of two >= x (for x >= 1)
    x--;
    x |= x >> 1;
    x |= x >> 2;
    x |= x >> 4;
    x |= x >> 8;
    x |= x >> 16;
    return x + 1;
}

int main(void) {
    printf("%d %d %d\n", popcount(0xF0F0), is_power_of_two(64), is_power_of_two(96));
    printf("%u %u\n", round_up_pow2(100), round_up_pow2(1024));
    uint8_t flags = 0;
    flags |= 1u << 3;                 // set bit 3
    printf("bit 3 set: %d, flags = 0x%02x\n", (flags >> 3) & 1, flags);
    return 0;
}
```

Compilers also offer built-ins (`__builtin_popcount` in GCC and Clang; C23 adds `<stdbit.h>` with `stdc_count_ones`).

#include <stdio.h>
#include <stdint.h>

typedef struct {
    uint16_t sensor;
    uint32_t time;
    int16_t centi_celsius;
} Reading;

void encode(const Reading *r, unsigned char out[9]) {
    uint16_t t = (uint16_t)r->centi_celsius; // two's-complement bits of the signed value
    out[0] = (unsigned char)(r->sensor >> 8);
    out[1] = (unsigned char)r->sensor;
    out[2] = (unsigned char)(r->time >> 24);
    out[3] = (unsigned char)(r->time >> 16);
    out[4] = (unsigned char)(r->time >> 8);
    out[5] = (unsigned char)r->time;
    out[6] = (unsigned char)(t >> 8);
    out[7] = (unsigned char)t;
    unsigned char sum = 0;
    for (int i = 0; i < 8; i++) sum ^= out[i];
    out[8] = sum;
}

int decode(const unsigned char in[9], Reading *r) {
    unsigned char sum = 0;
    for (int i = 0; i < 8; i++) sum ^= in[i];
    if (sum != in[8]) return -1;
    r->sensor = (uint16_t)((unsigned)in[0] << 8 | in[1]);
    r->time = (uint32_t)in[2] << 24 | (uint32_t)in[3] << 16 | (uint32_t)in[4] << 8 | in[5];
    r->centi_celsius = (int16_t)((unsigned)in[6] << 8 | in[7]);
    return 0;
}

int main(void) {
    Reading readings[] = {{7, 1773531392u, 2315}, {300, 1773531452u, -1000}, {65535, 0, 32767}};

## argc and argv

Command-line tools receive their arguments through `main`'s parameters: **`argc`** counts them and **`argv`** is an array of strings, with `argv[0]` the program's name and `argv[argc]` a `NULL` pointer:

```
$ ./grep -i -n pattern notes.txt
argc = 5
argv[0] = "./grep", argv[1] = "-i", argv[2] = "-n", argv[3] = "pattern", argv[4] = "notes.txt"
```

DevQuest runs programs without arguments, so the examples here pass their own `argv`-style array to a function that does the work, which is good design anyway: `main` stays tiny and the logic is testable.

```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

typedef struct {
    int ignore_case;
    int line_numbers;
    int count_only;
    const char *pattern;
    const char *file;
} Options;

/* Parses options like -i -n -c (also combined: -in); returns 0, or 2 on a usage error. */
int parse_args(int argc, char *argv[], Options *opt) {
    memset(opt, 0, sizeof *opt);
    int i = 1;
    for (; i < argc && argv[i][0] == '-' && argv[i][1] != '\0'; i++) {
        if (strcmp(argv[i], "--") == 0) {        // "--" ends the options
            i++;
            break;
        }
        for (const char *f = argv[i] + 1; *f; f++) {
            switch (*f) {
            case 'i': opt->ignore_case = 1; break;
            case 'n': opt->line_numbers = 1; break;
            case 'c': opt->count_only = 1; break;
            default:
                fprintf(stderr, "%s: unknown option -%c\n", argv[0], *f);
                return 2;
            }
        }
    }
    if (argc - i != 2) {
        printf("usage: %s [-icn] PATTERN FILE\n", argv[0]);
        return 2;
    }
    opt->pattern = argv[i];
    opt->file = argv[i + 1];
    return 0;
}

int main(void) {
    char *args[] = {"grep", "-in", "error", "app.log", NULL};
    Options opt;
    if (parse_args(4, args, &opt) == 0) {
        printf("pattern=%s file=%s ignore_case=%d line_numbers=%d count=%d\n",
               opt.pattern, opt.file, opt.ignore_case, opt.line_numbers, opt.count_only);
    }
    char *bad[] = {"grep", "error", NULL};
    return parse_args(2, bad, &opt) == 2 ? EXIT_SUCCESS : EXIT_FAILURE;
}
```

On POSIX systems, **`getopt`** (`<unistd.h>`) parses options in the standard way; it isn't part of standard C, so portable programs either write their own parser or use a small library.

## Conventions of well-behaved tools

- Read from **standard input** when no file is given, and write results to **standard output**, so the tool works in pipelines (`cat log | mytool | sort`).
- Write errors and diagnostics to **standard error**.
- **Exit status**: 0 for success, 1 for "ran fine but found nothing" or a failure, 2 for usage errors (as `grep` does). Scripts rely on it.
- `-h`/`--help` prints usage; unknown options are errors.
- Environment variables (`getenv("HOME")`, `getenv("NO_COLOR")`) carry configuration; `getenv` returns `NULL` when the variable isn't set.

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    const char *home = getenv("DEVQUEST_DOES_NOT_EXIST");
    printf("variable is %s\n", home ? home : "(not set)");
    const char *path = getenv("PATH");
    printf("PATH is %s\n", path && *path ? "set" : "not set");
    return 0;
}
```

#include <stdio.h>
#include <string.h>
#include <ctype.h>

int wc_main(int argc, char *argv[], const char *input) {
    int lines = 0, words = 0, bytes = 0;
    int i = 1;
    for (; i < argc && argv[i][0] == '-' && argv[i][1] != '\0'; i++) {
        for (const char *f = argv[i] + 1; *f; f++) {
            switch (*f) {
            case 'l': lines = 1; break;
            case 'w': words = 1; break;
            case 'c': bytes = 1; break;
            default:
                printf("wc: invalid option -- '%c'\n", *f);
                return 2;
            }
        }
    }
    if (!lines && !words && !bytes) lines = words = bytes = 1;
    const char *name = i < argc ? argv[i] : "-";

    int nl = 0, nw = 0, in_word = 0;
    size_t nc = strlen(input);
    for (const char *p = input; *p; p++) {
        if (*p == '\n') nl++;
        if (isspace((unsigned char)*p)) {
            in_word = 0;
        } else if (!in_word) {
            in_word = 1;
            nw++;
        }
    }
    if (lines) printf("%6d", nl);
    if (words) printf("%6d", nw);
    if (bytes) printf("%6zu", nc);
    printf(" %s\n", name);
    return 0;
}

int main(void) {
    const char *text = "the quick brown fox\njumps over\nthe lazy dog\n";
    char *a1[] = {"wc", "notes.txt", NULL};
    char *a2[] = {"wc", "-l", "notes.txt", NULL};
    char *a3[] = {"wc", "-wc", NULL};
    char *a4[] = {"wc", "-lz", "notes.txt", NULL};
    wc_main(2, a1, text);
    wc_main(3, a2, text);
    wc_main(2, a3, text);
    printf("exit %d\n", wc_main(3, a4, text));
    return 0;
}

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

static long freq[256];

static size_t rle_encode(const unsigned char *in, size_t n, unsigned char *out) {
    size_t o = 0;
    for (size_t i = 0; i < n;) {
        unsigned char b = in[i];
        size_t run = 1;
        while (i + run < n && in[i + run] == b && run < 255) run++;
        out[o++] = (unsigned char)run;
        out[o++] = b;
        i += run;
    }
    return o;
}

static size_t rle_decode(const unsigned char *in, size_t n, unsigned char *out) {
    size_t o = 0;
    for (size_t i = 0; i + 1 < n; i += 2) {
        for (unsigned c = 0; c < in[i]; c++) out[o++] = in[i + 1];
    }
    return o;
}

static int by_freq(const void *a, const void *b) {
    int x = *(const int *)a, y = *(const int *)b;
    if (freq[x] != freq[y]) return (freq[y] > freq[x]) - (freq[y] < freq[x]);
    return x - y;
}

static void print_hex(const unsigned char *p, size_t n) {
    for (size_t i = 0; i < n; i++) printf("%s%02X", i ? " " : "", p[i]);
}

int main(void) {
    char line[512];
    int number = 0;
    while (fgets(line, sizeof line, stdin) != NULL) {
        number++;
        unsigned char data[256], packed[512], back[256];
        size_t n = 0;
        int ok = 1;
        char *p = line;
        while (n < sizeof data) {
            while (*p == ' ' || *p == '\t') p++;
            if (*p == '\n' || *p == '\r' || *p == '\0') break;
            char *end;
            unsigned long v = strtoul(p, &end, 16);
            if (end == p || (*end != ' ' && *end != '\t' && *end != '\n' && *end != '\r' && *end != '\0') || v > 0xFF) {
                size_t len = strcspn(p, " \t\r\n");
                printf("line %d: bad byte \"%.*s\"\n", number, (int)len, p);
                ok = 0;
                break;
            }
            data[n++] = (unsigned char)v;
            p = end;
        }
        if (!ok || n == 0) continue;
        size_t packed_len = rle_encode(data, n, packed);
        size_t back_len = rle_decode(packed, packed_len, back);
        int same = back_len == n && memcmp(back, data, n) == 0;
        printf("line %d: ", number);
        print_hex(data, n);
        printf(" -> ");
        print_hex(packed, packed_len);
        printf(" (%s)\n", same ? "ok" : "MISMATCH");
        for (size_t i = 0; i < n; i++) freq[data[i]]++;
    }
    int order[256];
    for (int i = 0; i < 256; i++) order[i] = i;
    qsort(order, 256, sizeof order[0], by_freq);
    printf("top:");
    for (int i = 0; i < 3 && freq[order[i]] > 0; i++) {
        printf("%s %02X x%ld", i ? "," : "", order[i], freq[order[i]]);
    }
    printf("\n");
    return 0;
}
