## if, else if, else

```c
#include <stdio.h>

int main(void) {
    int temperature = 31;
    if (temperature > 30) {
        printf("hot\n");
    } else if (temperature > 15) {
        printf("mild\n");
    } else {
        printf("cold\n");
    }

    int stock = 0;
    if (!stock) printf("sold out\n"); // braces are optional for one statement...
    return 0;
}
```

The condition is any expression: 0 means false, anything else true. Braces are optional around a single statement, but **always write them**: adding a second line later without braces is a famous source of bugs (Apple's 2014 "goto fail" security hole was exactly that).

## switch

`switch` compares an **integer** expression (including `char` and `enum`) with constant cases:

```c
#include <stdio.h>

int main(void) {
    char op = '*';
    int a = 6, b = 7;
    switch (op) {
    case '+':
        printf("%d\n", a + b);
        break;
    case '*':
        printf("%d\n", a * b);
        break;
    case '/':
        if (b == 0) {
            printf("division by zero\n");
            break;
        }
        printf("%d\n", a / b);
        break;
    default:
        printf("unknown operator %c\n", op);
    }

    int month = 4, days;
    switch (month) {
    case 2:
        days = 28;
        break;
    case 4: case 6: case 9: case 11: // several labels share one block
        days = 30;
        break;
    default:
        days = 31;
    }
    printf("month %d has %d days\n", month, days);
    return 0;
}
```

**C's switch falls through**: without `break`, execution continues into the next case. Forgetting a `break` is a classic bug; when you fall through on purpose, say so with a comment (C23 adds `[[fallthrough]];`). A `switch` can't compare strings: use `strcmp` in an `if` chain.

=== lesson c-loops | Loops | 60
## while, do-while and for

```c
#include <stdio.h>

int main(void) {
    int n = 1;
    while (n < 100) {           // test first: may run zero times
        n *= 3;
    }
    printf("first power of 3 over 100: %d\n", n);

    int tries = 0;
    do {                        // test last: runs at least once
        tries++;
    } while (tries < 3);
    printf("tries: %d\n", tries);

    for (int i = 0; i < 5; i++) { // init; condition; step
        printf("%d ", i * i);
    }
    printf("\n");

    for (int i = 10, j = 0; i > j; i -= 3, j++) { // comma operator: several variables
        printf("(%d,%d) ", i, j);
    }
    printf("\n");
    return 0;
}
```

`for (;;)` and `while (1)` loop forever, until a `break` or `return`.

## break, continue and nested loops

```c
#include <stdio.h>

int main(void) {
    // print primes below 30
    for (int n = 2; n < 30; n++) {
        int prime = 1;
        for (int d = 2; d * d <= n; d++) {
            if (n % d == 0) {
                prime = 0;
                break;          // leaves only the inner loop
            }
        }
        if (!prime) continue;   // skip to the next n
        printf("%d ", n);
    }
    printf("\n");
    return 0;
}
```

C has no labelled `break`: to leave several loops, use a flag, put the loops in a function and `return`, or use `goto`.

## goto for clean-up

`goto` jumps to a label in the same function. It's rightly avoided for ordinary control flow, but one pattern is idiomatic in C (the Linux kernel uses it everywhere): **jumping to shared clean-up code** when an error happens halfway through acquiring resources:

```c
#include <stdio.h>
#include <stdlib.h>

int process(int fail_at) {
    int result = -1;
    int *a = NULL;
    int *b = NULL;
    a = malloc(100 * sizeof *a);
    if (a == NULL) goto out;
    b = malloc(100 * sizeof *b);
    if (b == NULL) goto out;
    if (fail_at == 2) {
        printf("step 2 failed\n");
        goto out;
    }
    printf("both buffers ready\n");
    result = 0;
out:
    free(b); // free(NULL) does nothing, so one label can clean up everything
    free(a);
    return result;
}

int main(void) {
    printf("result %d\n", process(0));
    printf("result %d\n", process(2));
    return 0;
}
```

Every failure jumps to the same clean-up code, which releases resources in reverse order. Bigger functions use several labels (`free_b:`, `free_a:`), each freeing what was acquired before that point. It's a structured way to avoid leaks without repeating the clean-up code in every error branch.

## Off-by-one errors

Arrays of `n` elements have indexes `0` to `n - 1`, so the standard loop is `for (int i = 0; i < n; i++)`. Writing `<=` reads one element past the end: undefined behaviour that might "work" for years. When counting down with an **unsigned** index, `i >= 0` is always true: a classic infinite loop.

## Declarations and definitions

A function must be **declared** before it's called. A **prototype** declares it (name, parameter types, return type) without the body; the **definition** can come later or in another file. Prototypes are what header files contain:

```c
#include <stdio.h>

double average(const int values[], int count); // prototype
void print_line(int width);

int main(void) {
    int scores[] = {80, 92, 75};
    print_line(20);
    printf("average: %.2f\n", average(scores, 3));
    print_line(20);
    return 0;
}

double average(const int values[], int count) {  // definition
    if (count == 0) return 0.0;
    int sum = 0;
    for (int i = 0; i < count; i++) {
        sum += values[i];
    }
    return (double)sum / count;
}

void print_line(int width) {                     // void: returns nothing
    for (int i = 0; i < width; i++) putchar('-');
    putchar('\n');
}
```

- `void` as a return type: the function returns nothing. `f(void)` in a declaration: it takes no arguments (in C before C23, `f()` means "unknown arguments", so always write `void`).
- An array parameter (`int values[]`) is really a **pointer** to the first element; the function can't know the length, so you pass it alongside. `const` promises not to modify the elements.

## Pass by value

Arguments are **copied** into the parameters. A function can't change the caller's variables, unless it receives their **addresses** (pointers):

```c
// warning
#include <stdio.h>

void try_to_double(int n) {
    n *= 2;                 // changes the local copy only
}

void really_double(int *n) {
    *n *= 2;                // changes the variable n points to
}

int main(void) {
    int x = 21;
    try_to_double(x);
    printf("%d\n", x);      // 21
    really_double(&x);
    printf("%d\n", x);      // 42
    return 0;
}
```

The compiler even warns that `try_to_double` sets its parameter without using it: a hint that the change is lost. That's also how C functions return several results: through pointer parameters, as `scanf` does.

## Scope and lifetime

- A **local** variable exists from its declaration to the end of its block `{ }`, and is created anew on every call (on the **stack**).
- A **global** variable (outside any function) lives for the whole program. Avoid them: any function can change them.
- A **`static` local** keeps its value between calls: it's initialised once and lives for the whole program.
- A **`static` function or global** is visible only in its own file: C's way to make something private.

```c
#include <stdio.h>

static int calls = 0;               // file-private global

int next_id(void) {
    static int id = 100;            // initialised once, keeps its value
    calls++;
    return id++;
}

int main(void) {
    printf("%d %d %d\n", next_id(), next_id(), next_id());
    printf("calls: %d\n", calls);
    return 0;
}
```

Look at the order of the three ids: C doesn't specify the order in which a function's arguments are evaluated, and many compilers evaluate them right to left, printing `102 101 100`. When the order of side effects matters, call the functions in separate statements.

## Recursion

A function can call itself; each call gets its own locals on the stack, and a **base case** stops the recursion. Too deep a recursion overflows the stack (usually a few megabytes):

```c
#include <stdio.h>

unsigned long long factorial(unsigned n) {
    return n <= 1 ? 1 : n * factorial(n - 1);
}

void hanoi(int disks, char from, char to, char via, int *moves) {
    if (disks == 0) return;
    hanoi(disks - 1, from, via, to, moves);
    (*moves)++;
    hanoi(disks - 1, via, to, from, moves);
}

int main(void) {
    printf("20! = %llu\n", factorial(20));
    int moves = 0;
    hanoi(10, 'A', 'C', 'B', &moves);
    printf("10 disks: %d moves\n", moves);
    return 0;
}
```

int gcd(int a, int b);

void stats(const int values[], int count, int *min, int *max, double *mean);

int main(void) {
    printf("gcd(84, 36) = %d\n", gcd(84, 36));
    int fractions[2][2] = {{12, -18}, {100, 25}};
    for (int i = 0; i < 2; i++) {
        int n = fractions[i][0], d = fractions[i][1];
        simplify(&n, &d);
        printf("%d/%d -> %d/%d\n", fractions[i][0], fractions[i][1], n, d);
    }
    int values[] = {12, 3, 42, 13};
    int lo, hi;
    double mean;
    stats(values, 4, &lo, &hi, &mean);
    printf("min %d, max %d, mean %.2f\n", lo, hi, mean);
    return 0;

int gcd(int a, int b) {
    if (a < 0) a = -a;
    if (b < 0) b = -b;
    return b == 0 ? a : gcd(b, a % b);
}

void simplify(int *num, int *den) {
    int g = gcd(*num, *den);
    *num /= g;
    *den /= g;
    if (*den < 0) {
        *num = -*num;
        *den = -*den;
    }
}

void stats(const int values[], int count, int *min, int *max, double *mean) {
    *min = *max = values[0];
    int sum = 0;
    for (int i = 0; i < count; i++) {
        if (values[i] < *min) *min = values[i];
        if (values[i] > *max) *max = values[i];
        sum += values[i];
    }
    *mean = (double)sum / count;
}

#include <stdio.h>

int is_prime(int n);
int digit_sum(int n);
int is_perfect(int n);
const char *parity(int n);

int main(void) {
    char line[64];
    int primes = 0;
    while (fgets(line, sizeof line, stdin) != NULL) {
        int n;
        if (sscanf(line, "%d", &n) != 1) continue;
        printf("%d: %s, digit sum %d", n, parity(n), digit_sum(n));
        if (is_prime(n)) {
            printf(", prime");
            primes++;
        }
        if (is_perfect(n)) printf(", perfect");
        printf("\n");
    }
    printf("primes: %d\n", primes);
    return 0;
}

int is_prime(int n) {
    if (n < 2) return 0;
    for (int d = 2; d * d <= n; d++) {
        if (n % d == 0) return 0;
    }
    return 1;
}

int digit_sum(int n) {
    if (n < 0) return digit_sum(-n);
    if (n < 10) return n;
    return n % 10 + digit_sum(n / 10);
}

int is_perfect(int n) {
    if (n < 2) return 0;
    int sum = 1;
    for (int d = 2; d * d <= n; d++) {
        if (n % d == 0) {
            sum += d;
            if (d != n / d) sum += n / d;
        }
    }
    return sum == n;
}

const char *parity(int n) {
    return n % 2 == 0 ? "even" : "odd";
}
