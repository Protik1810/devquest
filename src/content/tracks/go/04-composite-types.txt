## Arrays: fixed size

An **array** has a fixed length that is part of its type: `[3]int` and `[4]int` are different types. Arrays are **values**: assigning or passing one copies all its elements.

```go
package main

import "fmt"

func main() {
	var scores [3]int // [0 0 0]
	scores[0] = 90
	primes := [5]int{2, 3, 5, 7, 11}
	days := [...]string{"mon", "tue", "wed"} // length counted by the compiler
	fmt.Println(scores, len(primes), days[2])

	copyOf := primes // a full copy
	copyOf[0] = 100
	fmt.Println(primes[0], copyOf[0]) // 2 100
}
```

Arrays are used for fixed things (a 3×3 board, an RGB colour, a SHA-256 hash: `[32]byte`). Most of the time you use slices.

## Slices: flexible views

A **slice** (`[]int`, no length in the type) is a view of a section of an underlying array. It has a **length** (`len`, the elements it shows) and a **capacity** (`cap`, how far the underlying array extends).

```go
package main

import "fmt"

func main() {
	nums := []int{10, 20, 30, 40, 50} // a slice literal (creates the array for you)
	fmt.Println(nums, len(nums), cap(nums))

	middle := nums[1:4] // elements 1, 2, 3 (end excluded)
	fmt.Println(middle, len(middle), cap(middle))
	fmt.Println(nums[:2], nums[3:], nums[:]) // omitted bounds: start / end

	middle[0] = 99 // a slice shares the array: nums changes too!
	fmt.Println(nums)

	var empty []int // nil slice: len 0, ready to append to
	fmt.Println(empty == nil, len(empty))
	empty = append(empty, 1, 2, 3)
	fmt.Println(empty)

	buffer := make([]int, 3, 10) // length 3 (zeroed), capacity 10
	fmt.Println(buffer, len(buffer), cap(buffer))
}
```

## append

**`append(s, values...)`** adds elements at the end and returns the resulting slice, which you must assign back: `s = append(s, x)`. When there's no capacity left, append allocates a **bigger array** (roughly doubling), copies the elements, and the new slice points there:

```go
package main

import "fmt"

func main() {
	var s []int
	prevCap := -1
	for i := range 20 {
		s = append(s, i)
		if cap(s) != prevCap {
			fmt.Printf("len %2d cap %2d\n", len(s), cap(s))
			prevCap = cap(s)
		}
	}
	more := []int{100, 200}
	s = append(s, more...) // append a whole slice
	fmt.Println(len(s), s[len(s)-1])
}
```

When you know the final size, `make([]T, 0, n)` allocates once instead of several times.

package main

import "fmt"

func chunk(items []string, size int) [][]string {
	var groups [][]string
	for i := 0; i < len(items); i += size {
		end := min(i+size, len(items))
		groups = append(groups, items[i:end])
	}
	return groups
}

func rotate(nums []int, k int) []int {
	if len(nums) == 0 {
		return nil
	}
	k %= len(nums)
	out := make([]int, 0, len(nums))
	out = append(out, nums[k:]...)
	return append(out, nums[:k]...)
}

func main() {
	letters := []string{"a", "b", "c", "d", "e", "f", "g"}
	fmt.Println(chunk(letters, 3))
	fmt.Println(chunk(letters, 10))
	nums := []int{1, 2, 3, 4, 5}
	fmt.Println(rotate(nums, 2))
	fmt.Println(rotate(nums, 11))
	fmt.Println("original:", nums)
}

## The sharing trap

Because slices share arrays, an `append` to one slice can **overwrite** data visible through another, when there's spare capacity:

```go
package main

import "fmt"

func main() {
	base := make([]int, 3, 10) // [0 0 0], room for 10
	a := append(base, 1)       // uses base's spare capacity
	b := append(base, 2)       // ... and so does b: same array, same position!
	fmt.Println(a, b)          // [0 0 0 2] [0 0 0 2]

	// the fix: copy when you need an independent slice
	c := append([]int(nil), base...) // or slices.Clone(base)
	c = append(c, 1)
	d := cloneInts(base)
	d = append(d, 2)
	fmt.Println(c, d)
}

func cloneInts(s []int) []int {
	out := make([]int, len(s))
	copy(out, s) // copy(dst, src) copies min(len(dst), len(src)) elements
	return out
}
```

Rules of thumb: a function that receives a slice may change its **elements** (the caller sees it); if it appends, it must **return** the new slice; and when you keep a slice for later, copy it if the caller might reuse their array.

## The slices package

Since Go 1.21 the standard library has generic helpers in `slices` (and `maps`) that replace many hand-written loops:

```go
package main

import (
	"fmt"
	"slices"
	"strings"
)

func main() {
	nums := []int{5, 2, 8, 2, 9, 1}
	sorted := slices.Clone(nums)
	slices.Sort(sorted)
	fmt.Println(sorted, nums)
	fmt.Println(slices.Contains(nums, 8), slices.Index(nums, 2), slices.Max(nums), slices.Min(nums))

	i, found := slices.BinarySearch(sorted, 8)
	fmt.Println(i, found)
	fmt.Println(slices.Compact(sorted)) // removes consecutive duplicates
	fmt.Println(slices.Equal([]int{1, 2}, []int{1, 2}))

	words := []string{"pear", "Fig", "apple", "kiwi"}
	slices.SortFunc(words, func(a, b string) int {
		return strings.Compare(strings.ToLower(a), strings.ToLower(b))
	})
	fmt.Println(words)

	nums = slices.Insert(nums, 1, 100)
	nums = slices.Delete(nums, 3, 5) // removes elements 3 and 4
	fmt.Println(nums)
}
```

Sorting by several keys uses `cmp.Compare` and `cmp.Or`, which you'll use with structs in a moment.

## Removing while iterating, and filtering in place

A common idiom filters a slice **without allocating**, by reusing its array:

```go
package main

import (
	"fmt"
	"slices"
)

func main() {
	nums := []int{1, 2, 3, 4, 5, 6, 7, 8}
	kept := nums[:0] // same array, length 0
	for _, n := range nums {
		if n%2 == 0 {
			kept = append(kept, n)
		}
	}
	fmt.Println(kept)

	other := []int{3, 1, 4, 1, 5, 9, 2, 6}
	other = slices.DeleteFunc(other, func(n int) bool { return n < 4 })
	fmt.Println(other)
}
```

package main

	"fmt"
	"slices"

func topN(scores []int, n int) []int {
	sorted := slices.Clone(scores)
	slices.Sort(sorted)
	slices.Reverse(sorted)
	return sorted[:min(n, len(sorted))]
}

func dedupe(names []string) []string {
	out := slices.Clone(names)
	slices.Sort(out)
	return slices.Compact(out)

	fmt.Println("unique:", dedupe(names))
	fmt.Println("names unchanged:", names)

## Key-value lookup

A **map** associates keys with values: `map[string]int` maps strings to ints. Lookups, insertions and deletions are fast (a hash table) regardless of size.

```go
package main

import (
	"fmt"
	"maps"
	"slices"
)

func main() {
	stock := map[string]int{"apple": 5, "pear": 0}
	stock["kiwi"] = 12 // insert or update
	stock["apple"] += 3
	fmt.Println(stock["apple"], len(stock))

	fmt.Println(stock["mango"]) // a missing key gives the zero value: 0

	n, ok := stock["pear"] // "comma ok": was the key there?
	fmt.Println(n, ok)
	_, ok = stock["mango"]
	fmt.Println(ok)

	delete(stock, "pear") // no error if the key is missing
	fmt.Println(stock)    // fmt prints maps sorted by key

	keys := slices.Sorted(maps.Keys(stock)) // sorted keys for a stable order
	for _, k := range keys {
		fmt.Println(k, stock[k])
	}

	counts := make(map[string]int) // make an empty map
	for _, w := range []string{"go", "is", "go", "fun", "go"} {
		counts[w]++ // the zero value makes counting easy
	}
	fmt.Println(counts)
}
```

- Keys must be **comparable** (`==` works): strings, numbers, booleans, arrays, and structs of those. Slices, maps and functions can't be keys.
- A **nil map** (`var m map[string]int`) can be read, but writing to it **panics**. Create maps with a literal or `make`.
- Iteration order is randomised; `fmt` prints maps sorted, but `range` doesn't.
- Like slices, maps are references: passing a map to a function lets it change the map.

## Sets and grouping

Go has no set type: a `map[T]bool` (or `map[T]struct{}`, which uses no memory for values) does the job:

```go
package main

import (
	"fmt"
	"maps"
	"slices"
	"strings"
)

func main() {
	seen := map[string]bool{}
	for _, tag := range []string{"go", "web", "go", "cli", "web"} {
		seen[tag] = true
	}
	fmt.Println(len(seen), seen["cli"], seen["rust"])

	// group words by their first letter
	groups := map[byte][]string{}
	for _, w := range strings.Fields("apple avocado banana blueberry cherry") {
		groups[w[0]] = append(groups[w[0]], w)
	}
	for _, k := range slices.Sorted(maps.Keys(groups)) {
		fmt.Printf("%c: %v\n", k, groups[k])
	}

	// nested maps need the inner map created first
	scores := map[string]map[string]int{}
	add := func(student, subject string, s int) {
		if scores[student] == nil {
			scores[student] = map[string]int{}
		}
		scores[student][subject] = s
	}
	add("ada", "math", 95)
	add("ada", "art", 80)
	fmt.Println(scores)
}
```

package main

import (
	"cmp"
	"fmt"
	"maps"
	"slices"
	"strings"
)

func wordStats(text string) (map[string]int, []string) {
	counts := map[string]int{}
	for _, w := range strings.Fields(text) {
		w = strings.ToLower(strings.Trim(w, ".,!?"))
		if w != "" {
			counts[w]++
		}
	}
	order := slices.Collect(maps.Keys(counts))
	slices.SortFunc(order, func(a, b string) int {
		return cmp.Or(cmp.Compare(counts[b], counts[a]), cmp.Compare(a, b))
	})
	return counts, order
}

	counts, order := wordStats(text)
	for _, w := range order[:3] {

	fmt.Println("distinct:", len(counts))

## Grouping data

A **struct** groups named fields into one type. It's how Go models things: users, orders, points.

```go
package main

import "fmt"

type Point struct {
	X, Y float64
}

type User struct {
	Name   string
	Email  string
	Age    int
	Admin  bool
	Tags   []string
}

func main() {
	p := Point{X: 1.5, Y: -2}
	q := Point{3, 4} // positional: fragile, avoid for structs with many fields
	var origin Point // zero value: {0 0}
	fmt.Println(p, q, origin, p.X)

	u := User{Name: "Ada", Email: "ada@example.com"} // missing fields get zero values
	u.Age = 36
	fmt.Printf("%v\n%+v\n", u, u) // %+v shows field names

	copied := u // structs are values: this copies every field
	copied.Name = "Grace"
	fmt.Println(u.Name, copied.Name)
	fmt.Println(p == Point{1.5, -2}) // structs compare field by field

	pair := struct{ Key, Value string }{"lang", "go"} // anonymous struct
	fmt.Println(pair.Key, pair.Value)
}
```

Fields starting with a capital letter are exported, exactly like functions. A struct with only comparable fields can be compared with `==` and used as a map key.

## Constructors

Go has no constructors; by convention, a function **`NewThing`** creates a valid value, and the zero value should be useful when possible:

```go
package main

import (
	"errors"
	"fmt"
	"time"
)

type Task struct {
	Title   string
	Due     time.Time
	Done    bool
	Created time.Time
}

func NewTask(title string, due time.Time) (Task, error) {
	if title == "" {
		return Task{}, errors.New("title is required")
	}
	return Task{Title: title, Due: due, Created: time.Date(2026, 3, 1, 9, 0, 0, 0, time.UTC)}, nil
}

func main() {
	t, err := NewTask("write report", time.Date(2026, 3, 5, 17, 0, 0, 0, time.UTC))
	fmt.Println(t.Title, t.Due.Format("2006-01-02"), t.Done, err)
	_, err = NewTask("", time.Time{})
	fmt.Println(err)
}
```

## Embedding: composition instead of inheritance

Go has no inheritance. Instead, a struct can **embed** another type: the embedded type's fields (and methods, next module) are **promoted**, so they can be used as if they belonged to the outer struct. It's composition with convenient access:

```go
package main

import "fmt"

type Address struct {
	Street, City string
}

type Contact struct {
	Email string
	Phone string
}

type Customer struct {
	Name    string
	Address // embedded: no field name
	Contact
	Billing Address // a normal named field
}

func main() {
	c := Customer{
		Name:    "Ada",
		Address: Address{"1 Main St", "London"},
		Contact: Contact{Email: "ada@example.com"},
		Billing: Address{City: "Cambridge"},
	}
	fmt.Println(c.City, c.Email)         // promoted fields
	fmt.Println(c.Address.City)           // the long form still works
	fmt.Println(c.Billing.City)
}
```

A `Customer` *has* an address; it isn't a kind of address. If two embedded types have a field with the same name, you must use the long form.

=== lesson go-pointers | Pointers | 65
## Addresses

A **pointer** holds the memory address of a value. `&x` takes the address of `x`; `*p` reads or writes the value at that address. The type of a pointer to an `int` is `*int`.

```go
package main

import "fmt"

func main() {
	x := 10
	p := &x            // p points to x
	fmt.Println(*p)    // 10: read through the pointer
	*p = 20            // write through it
	fmt.Println(x)     // 20
	fmt.Println(p != nil)

	var q *int         // the zero value of a pointer is nil
	fmt.Println(q == nil)
	// fmt.Println(*q) would panic: nil pointer dereference

	n := new(int)      // allocates a zero int and returns its address
	*n = 5
	fmt.Println(*n)
}
```

Go has **no pointer arithmetic** (you can't do `p + 1` like in C), and memory is garbage collected: returning the address of a local variable is perfectly safe, the value lives as long as something points to it.

## Why pointers matter: sharing and changing

Everything in Go is passed by value. To let a function **change** a variable, pass a pointer to it. Pointers also avoid copying large structs:

```go
package main

import "fmt"

type Account struct {
	Owner   string
	Balance int
}

func depositValue(a Account, amount int) {
	a.Balance += amount // changes a copy: the caller sees nothing
}

func deposit(a *Account, amount int) {
	a.Balance += amount // a.Balance is shorthand for (*a).Balance
}

func NewAccount(owner string) *Account {
	return &Account{Owner: owner} // safe: escapes to the heap
}

func main() {
	acc := Account{Owner: "Ada"}
	depositValue(acc, 50)
	fmt.Println(acc.Balance) // 0
	deposit(&acc, 50)
	fmt.Println(acc.Balance) // 50

	b := NewAccount("Bob")
	deposit(b, 30)
	alias := b // copies the pointer, not the account
	alias.Balance += 1
	fmt.Println(b.Owner, b.Balance)
}
```

Go automatically dereferences pointers to structs when you access fields (`a.Balance`), so the syntax stays light.

## Slices and maps are already references (sort of)

A slice is a small struct (pointer to the array, length, capacity) and a map is a pointer to a hash table, so passing them copies only that small header: the function sees the **same elements**. But a function that `append`s changes only its copy of the header, which is why append's result must be returned.

```go
package main

import "fmt"

func double(nums []int) {
	for i := range nums {
		nums[i] *= 2 // visible to the caller
	}
}

func addOne(nums []int) {
	nums = append(nums, 1) // only the local header changes
}

func main() {
	s := []int{1, 2, 3}
	double(s)
	addOne(s)
	fmt.Println(s) // [2 4 6]
}
```

## When to use a pointer

- the function must **modify** the value;
- the struct is **large** and copied often;
- you need "**no value**" (`nil`), for example an optional field;
- the value must not be copied, because it contains a mutex or another value that identifies itself (Module 6).

Small structs that represent values (points, colours, money) are usually passed by value.

package main

import "fmt"

type Product struct {
	Name  string
	Cents int
}

func swap(a, b *int) {
	*a, *b = *b, *a
}

func applyDiscount(items []*Product, percent int) {
	for _, p := range items {
		p.Cents -= p.Cents * percent / 100
	}
}

func cheapest(items []*Product) *Product {
	var best *Product
	for _, p := range items {
		if best == nil || p.Cents < best.Cents {
			best = p
		}
	}
	return best
}

func main() {
	x, y := 1, 2
	swap(&x, &y)
	fmt.Printf("x=%d y=%d\n", x, y)

		"lamp": {"lamp", 3500},
		"desk": {"desk", 15000},

	fmt.Println(list[0].Name, list[0].Cents, list[1].Name, list[1].Cents)
	fmt.Println("catalogue lamp:", catalogue["lamp"].Cents)

::solution
package main

import (
	"cmp"
	"fmt"
	"maps"
	"slices"
)

type Student struct {
	Name   string
	Scores map[string]int
}

type Gradebook struct {
	students map[string]*Student
}

func NewGradebook() *Gradebook {
	return &Gradebook{students: map[string]*Student{}}
}

func addScore(gb *Gradebook, name, subject string, score int) {
	s, ok := gb.students[name]
	if !ok {
		s = &Student{Name: name, Scores: map[string]int{}}
		gb.students[name] = s
	}
	s.Scores[subject] = score
}

func average(s *Student) float64 {
	if len(s.Scores) == 0 {
		return 0
	}
	total := 0
	for _, v := range s.Scores {
		total += v
	}
	return float64(total) / float64(len(s.Scores))
}

func ranking(gb *Gradebook) []*Student {
	list := slices.Collect(maps.Values(gb.students))
	slices.SortFunc(list, func(a, b *Student) int {
		return cmp.Or(cmp.Compare(average(b), average(a)), cmp.Compare(a.Name, b.Name))
	})
	return list
}

func subjectBest(gb *Gradebook) map[string]string {
	best := map[string]string{}
	bestScore := map[string]int{}
	for _, name := range slices.Sorted(maps.Keys(gb.students)) {
		for subject, score := range gb.students[name].Scores {
			if _, seen := best[subject]; !seen || score > bestScore[subject] {
				best[subject], bestScore[subject] = name, score
			}
		}
	}
	return best
}

func main() {
	gb := NewGradebook()
	addScore(gb, "Ada", "math", 90)
	addScore(gb, "Ada", "art", 85)
	addScore(gb, "Bob", "math", 70)
	addScore(gb, "Bob", "music", 78)
	addScore(gb, "Cy", "math", 95)
	addScore(gb, "Cy", "art", 85)
	addScore(gb, "Cy", "music", 93)
	addScore(gb, "Bob", "music", 78)
	for i, s := range ranking(gb) {
		fmt.Printf("%d. %s %.1f\n", i+1, s.Name, average(s))
	}
	best := subjectBest(gb)
	for _, subject := range slices.Sorted(maps.Keys(best)) {
		fmt.Printf("%s: %s\n", subject, best[subject])
	}
}
