=== module Capstone projects

=== lesson go-project | Capstone 1: an inventory command-line tool | 400
Go is famous for command-line tools, so the first capstone builds one in two steps: first the **domain** (products, stock, sales and errors), then a **command interpreter** that reads commands from standard input and prints a JSON report. Step 2's starter contains step 1's solution, so you can continue even if step 1 gave you trouble.

## How a task like this is approached

1. **Model the domain** before the user interface: a `Product` is plain data, the `Inventory` owns the state and enforces the rules, and prices are **integer cents** (never floats for money).
2. **Make errors part of the design**: a sentinel `ErrNotFound` for "no such product" and a custom `*StockError` that tells the caller how much is left, so the interface can phrase its own message with `errors.Is` and `errors.As`.
3. **Keep parsing separate from logic**: the interpreter turns text into calls and formats results; the inventory knows nothing about text. Each part is easy to test alone.
4. **Report with encoding/json**: a separate report type with struct tags keeps the JSON format independent of the internal types.

`main` is the specification: its printed output must match exactly. Don't change it.
--- goals
- model a domain with structs, maps of pointers and integer cents
- design errors with a sentinel value and a custom error type
- parse line-based commands robustly and report every problem with its line number
- produce a JSON report with dedicated types and struct tags

**Step 1: the inventory.** `Product`, `ErrNotFound` and `StockError` are given. Complete the `Inventory` methods:

- `Add(sku, name, priceCents, qty)`: errors `add: sku is required`, `add <sku>: price must be positive`, `add <sku>: quantity can't be negative` (checked in that order); adding an existing SKU updates its name and price and **adds** to its quantity;
- `Sell(sku, qty)` returns the revenue in cents; an unknown SKU returns `sell <sku>: not found` wrapping `ErrNotFound` (`%w`); too little stock returns `&StockError{SKU, Have, Want}` and changes nothing; a sale adds to the total revenue;
- `Restock(sku, qty)`: `restock <sku>: quantity must be positive` for qty ≤ 0, `restock <sku>: not found` (wrapping `ErrNotFound`) for an unknown SKU;
- `LowStock(threshold)` returns copies of the products with at most `threshold` units, fewest first, ties by SKU (sort the SKUs first, then `slices.SortStableFunc` by quantity).

	check(inv.Add("", "Mystery", 100, 1))
	check(inv.Add("MUG", "Mug", 0, 5))

		sku string
		qty int
	}{{"PEN-1", 30}, {"LAMP", 5}, {"NB-A5", 12}, {"DESK", 1}, {"LAMP", 2}} {

func Money(cents int) string { return fmt.Sprintf("%d.%02d", cents/100, cents%100) }

type Product struct {
	SKU        string
	Name       string
	PriceCents int
	Qty        int
}

var ErrNotFound = errors.New("not found")

type StockError struct {
	SKU        string
	Have, Want int
}

func (e *StockError) Error() string {
	return fmt.Sprintf("not enough %s: have %d, want %d", e.SKU, e.Have, e.Want)
}

type Inventory struct {
	products map[string]*Product
	revenue  int
}

func NewInventory() *Inventory {
	return &Inventory{products: map[string]*Product{}}
}

func (inv *Inventory) Add(sku, name string, priceCents, qty int) error {
	switch {
	case sku == "":
		return errors.New("add: sku is required")
	case priceCents <= 0:
		return fmt.Errorf("add %s: price must be positive", sku)
	case qty < 0:
		return fmt.Errorf("add %s: quantity can't be negative", sku)
	}
	if p, ok := inv.products[sku]; ok {
		p.Name, p.PriceCents = name, priceCents
		p.Qty += qty
		return nil
	}
	inv.products[sku] = &Product{sku, name, priceCents, qty}
	return nil
}

func (inv *Inventory) Sell(sku string, qty int) (int, error) {
	p, ok := inv.products[sku]
	if !ok {
		return 0, fmt.Errorf("sell %s: %w", sku, ErrNotFound)
	}
	if qty > p.Qty {
		return 0, &StockError{SKU: sku, Have: p.Qty, Want: qty}
	}
	p.Qty -= qty
	revenue := qty * p.PriceCents
	inv.revenue += revenue
	return revenue, nil
}

func (inv *Inventory) Restock(sku string, qty int) error {
	if qty <= 0 {
		return fmt.Errorf("restock %s: quantity must be positive", sku)
	}
	p, ok := inv.products[sku]
	if !ok {
		return fmt.Errorf("restock %s: %w", sku, ErrNotFound)
	}
	p.Qty += qty
	return nil
}

func (inv *Inventory) LowStock(threshold int) []Product {
	var low []Product
	for _, sku := range slices.Sorted(maps.Keys(inv.products)) {
		if p := inv.products[sku]; p.Qty <= threshold {
			low = append(low, *p)
		}
	}
	slices.SortStableFunc(low, func(a, b Product) int { return cmp.Compare(a.Qty, b.Qty) })
	return low
}

func (inv *Inventory) Revenue() int { return inv.revenue }

import (
	"bufio"
	"cmp"
	"encoding/json"
	"errors"
	"fmt"
	"maps"
	"os"
	"slices"
	"strconv"
	"strings"
)

func main() {
	inv := NewInventory()
	in := bufio.NewScanner(os.Stdin)
	for line := 1; in.Scan(); line++ {
		text := strings.TrimSpace(in.Text())
		if text == "" || strings.HasPrefix(text, "#") {
			continue
		}
		out, err := Execute(inv, text)
		if err != nil {
			fmt.Printf("line %d: error: %v\n", line, err)
			continue
		}
		if out != "" {
			fmt.Println(out)
		}
	}
}

// Execute runs one command against the inventory and returns what to print.
func Execute(inv *Inventory, command string) (string, error) {
	f := strings.Fields(command)
	need := func(n int, usage string) error {
		if len(f) != n {
			return fmt.Errorf("usage: %s", usage)
		}
		return nil
	}
	switch f[0] {
	case "add":
		if err := need(5, "add <sku> <name> <price> <qty>"); err != nil {
			return "", err
		}
		price, err := ParseCents(f[3])
		if err != nil {
			return "", err
		}
		qty, err := strconv.Atoi(f[4])
		if err != nil {
			return "", fmt.Errorf("bad quantity %q", f[4])
		}
		return "", inv.Add(f[1], strings.ReplaceAll(f[2], "_", " "), price, qty)
	case "sell", "restock":
		if err := need(3, f[0]+" <sku> <qty>"); err != nil {
			return "", err
		}
		qty, err := strconv.Atoi(f[2])
		if err != nil || qty <= 0 {
			return "", fmt.Errorf("bad quantity %q", f[2])
		}
		if f[0] == "restock" {
			return "", inv.Restock(f[1], qty)
		}
		revenue, err := inv.Sell(f[1], qty)
		if err != nil {
			return "", err
		}
		return fmt.Sprintf("sold %d %s for %s", qty, f[1], Money(revenue)), nil
	case "low":
		if err := need(2, "low <threshold>"); err != nil {
			return "", err
		}
		n, err := strconv.Atoi(f[1])
		if err != nil {
			return "", fmt.Errorf("bad threshold %q", f[1])
		}
		var parts []string
		for _, p := range inv.LowStock(n) {
			parts = append(parts, fmt.Sprintf("%s(%d)", p.SKU, p.Qty))
		}
		return "low stock: " + strings.Join(parts, ", "), nil
	case "report":
		return Report(inv)
	}
	return "", fmt.Errorf("unknown command %q", f[0])
}

// ParseCents turns "12", "12.5" or "12.50" into cents.
func ParseCents(s string) (int, error) {
	whole, frac, _ := strings.Cut(s, ".")
	if len(frac) > 2 {
		return 0, fmt.Errorf("bad price %q", s)
	}
	frac += strings.Repeat("0", 2-len(frac))
	euros, err1 := strconv.Atoi(whole)
	cents, err2 := strconv.Atoi(frac)
	if err1 != nil || err2 != nil || euros < 0 {
		return 0, fmt.Errorf("bad price %q", s)
	}
	return euros*100 + cents, nil
}

type reportLine struct {
	SKU   string `json:"sku"`
	Name  string `json:"name"`
	Price string `json:"price"`
	Qty   int    `json:"qty"`
	Value string `json:"stock_value"`
}

type report struct {
	Products []reportLine `json:"products"`
	Revenue  string       `json:"revenue"`
}

// Report returns the inventory as indented JSON, products sorted by SKU.
func Report(inv *Inventory) (string, error) {
	r := report{Products: []reportLine{}, Revenue: Money(inv.Revenue())}
	for _, sku := range slices.Sorted(maps.Keys(inv.products)) {
		p := inv.products[sku]
		r.Products = append(r.Products, reportLine{p.SKU, p.Name, Money(p.PriceCents), p.Qty, Money(p.PriceCents * p.Qty)})
	}
	data, err := json.MarshalIndent(r, "", "  ")
	return string(data), err
}

=== lesson go-capstone-concurrent | Capstone 2: a concurrent search index | 450
Search engines build an **inverted index**: for each word, the documents that contain it and how often. This capstone indexes a folder of text files and answers multi-word queries, first one file at a time, then with a **pool of goroutines** that stops cleanly at the first broken file.

## The design

- `Tokenize(text)` turns text into lower-case words (letters and digits only), so "Go," and "go" are the same word.
- The index is a `map[string]map[string]int`: word → document → count.
- A query's words must **all** appear in a document (AND); its score is the total count of the query words, and results are sorted by score, then by name.
- `indexFile` (given) reads one file; it simulates a slow disk (20 ms) and refuses binary files with `ErrBinary`. It also records how many files are being read at the same time, so the harness can tell whether your indexing really runs concurrently.

## From sequential to concurrent

Step 1 reads the files one after another: correct, but each file waits for the previous one. Step 2 keeps the same results and adds **concurrency**:

1. a feeder goroutine sends the file paths into a channel (stopping when the context is cancelled);
2. `workers` goroutines receive paths, index the files and merge them into the shared index, protected by a **mutex**;
3. the first error is recorded once and **cancels** the context, so the remaining work stops;
4. `BuildIndex` waits for the workers and returns the index or the first error, wrapped.
--- goals
- build an inverted index with nested maps and answer AND queries with ranking
- tokenize text with strings.FieldsFunc and unicode
- index files concurrently with a worker pool, a mutex and a WaitGroup
- stop all workers at the first error with context cancellation

// BuildIndex indexes every .txt file of dir, one after another, and stops at the first error.
func BuildIndex(dir string) (*Index, error) {

	if err != nil {
		return nil, err

	for _, p := range paths {
		counts, err := indexFile(p)
		if err != nil {
			return nil, fmt.Errorf("build index: %w", err)
		}
		idx.add(filepath.Base(p), counts)
	}
	return idx, nil

// Search returns the documents containing every word of the query,
// scored by the total number of occurrences, best first, then by name.
func (idx *Index) Search(query string) []Hit {
	words := Tokenize(query)
	if len(words) == 0 {
		return nil
	}
	scores := map[string]int{}
	for doc, n := range idx.words[words[0]] {
		scores[doc] = n
	}
	for _, w := range words[1:] {
		for doc := range scores {
			n, ok := idx.words[w][doc]
			if !ok {
				delete(scores, doc)
				continue
			}
			scores[doc] += n
		}
	}
	var hits []Hit
	for doc, s := range scores {
		hits = append(hits, Hit{doc, s})
	}
	slices.SortFunc(hits, func(a, b Hit) int {
		return cmp.Or(cmp.Compare(b.Score, a.Score), cmp.Compare(a.Doc, b.Doc))
	})
	return hits
}

func (idx *Index) add(doc string, counts map[string]int) {
	idx.mu.Lock()
	defer idx.mu.Unlock()
	for w, n := range counts {
		if idx.words[w] == nil {
			idx.words[w] = map[string]int{}
		}
		idx.words[w][doc] = n
	}
	idx.files++
}

// BuildIndex indexes every .txt file of dir with a pool of workers and stops at the first error.
func BuildIndex(ctx context.Context, dir string, workers int) (*Index, error) {
	paths, err := filepath.Glob(filepath.Join(dir, "*.txt"))
	if err != nil {
		return nil, err
	}
	ctx, cancel := context.WithCancel(ctx)
	defer cancel()

	jobs := make(chan string)
	go func() {
		defer close(jobs)
		for _, p := range paths {
			select {
			case jobs <- p:
			case <-ctx.Done():
				return
			}
		}
	}()

	idx := &Index{words: map[string]map[string]int{}}
	var once sync.Once
	var firstErr error
	var wg sync.WaitGroup
	for range workers {
		wg.Go(func() {
			for p := range jobs {
				if ctx.Err() != nil {
					continue
				}
				counts, err := indexFile(p)
				if err != nil {
					once.Do(func() { firstErr = err })
					cancel()
					continue
				}
				idx.add(filepath.Base(p), counts)
			}
		})
	}
	wg.Wait()
	if firstErr != nil {
		return nil, fmt.Errorf("build index: %w", firstErr)
	}
	return idx, nil
}

// Search returns the documents containing every word of the query,
// scored by the total number of occurrences, best first, then by name.
func (idx *Index) Search(query string) []Hit {
	words := Tokenize(query)
	if len(words) == 0 {
		return nil
	}
	scores := map[string]int{}
	for doc, n := range idx.words[words[0]] {
		scores[doc] = n
	}
	for _, w := range words[1:] {
		for doc := range scores {
			n, ok := idx.words[w][doc]
			if !ok {
				delete(scores, doc)
				continue
			}
			scores[doc] += n
		}
	}
	var hits []Hit
	for doc, s := range scores {
		hits = append(hits, Hit{doc, s})
	}
	slices.SortFunc(hits, func(a, b Hit) int {
		return cmp.Or(cmp.Compare(b.Score, a.Score), cmp.Compare(a.Doc, b.Doc))
	})
	return hits
}

=== lesson go-capstone-solo | Capstone 3: build it from a spec | 450
The final capstone gives you a **specification** and a test harness, and nothing else: you design and write the types from scratch. You'll build a tiny in-memory **key-value store** in the spirit of Redis: values with expiry times, counters and **nested transactions**.

## The specification

**Types and functions** (names fixed by the harness):

- `ErrNotInteger` (`value is not an integer`) and `ErrNoTransaction` (`no transaction`), sentinel errors;
- `type Clock interface { Now() time.Time }` (the harness passes a fake clock it advances by hand);
- `type Store struct` and `func NewStore(clock Clock) *Store`, with the methods below.

**Rules:**

1. `Set(key, value)` stores a value without expiry; `SetTTL(key, value, ttl)` stores one that expires `ttl` after the clock's current time. A key is **expired** once the clock reaches its expiry time; expired keys behave exactly like missing ones everywhere.
2. `Get(key) (string, bool)`; `Delete(key) bool` (false when the key is missing or expired).
3. `Incr(key) (int, error)`: a missing key counts as 0; a value that isn't an integer returns `incr <key>: value is not an integer` wrapping `ErrNotInteger`; the expiry of an existing key is kept.
4. `TTL(key) (time.Duration, bool)`: the remaining time, or false when the key is missing, expired or has no expiry.
5. `Keys(prefix) []string`: the live keys starting with the prefix, sorted (an empty slice, not nil, when there are none).
6. `Begin()` starts a transaction; transactions can be **nested**. Changes inside a transaction are visible immediately. `Rollback()` undoes every change made since the matching `Begin` (including deletes and new keys); `Commit()` keeps the changes, but if an outer transaction is open, rolling **it** back must still undo them. Both return `ErrNoTransaction` when no transaction is open. `Depth()` returns the number of open transactions.

## How to approach it

1. Write the types and method signatures first, with placeholder bodies, so the harness compiles.
2. Store entries as a value plus an expiry `time.Time` (the zero time meaning "never"); one helper `live(key)` that hides expired entries keeps every method simple.
3. For transactions, keep a **stack of undo logs**: the first time a transaction changes a key, remember whether the key existed and its old entry. Rollback restores them; Commit hands the inner log to the outer transaction (keeping the outer's own records, which are older).
4. Run the harness and compare the output line by line.
--- goals
- design a set of types from a written specification
- inject a clock through an interface to make time testable
- implement expiry consistently through one helper
- implement nested transactions with a stack of undo logs

var (
	ErrNotInteger    = errors.New("value is not an integer")
	ErrNoTransaction = errors.New("no transaction")
)

type Clock interface{ Now() time.Time }

type entry struct {
	value   string
	expires time.Time // zero: never
}

// undo remembers a key's state before the current transaction first changed it.
type undo struct {
	existed bool
	old     entry
}

type Store struct {
	clock Clock
	data  map[string]entry
	txs   []map[string]undo
}

func NewStore(clock Clock) *Store {
	return &Store{clock: clock, data: map[string]entry{}}
}

func (s *Store) live(key string) (entry, bool) {
	e, ok := s.data[key]
	if !ok || (!e.expires.IsZero() && !s.clock.Now().Before(e.expires)) {
		return entry{}, false
	}
	return e, true
}

// remember records the key's state the first time the innermost transaction touches it.
func (s *Store) remember(key string) {
	if len(s.txs) == 0 {
		return
	}
	tx := s.txs[len(s.txs)-1]
	if _, seen := tx[key]; seen {
		return
	}
	e, ok := s.live(key)
	tx[key] = undo{ok, e}
}

func (s *Store) Set(key, value string) { s.SetTTL(key, value, 0) }

func (s *Store) SetTTL(key, value string, ttl time.Duration) {
	s.remember(key)
	e := entry{value: value}
	if ttl > 0 {
		e.expires = s.clock.Now().Add(ttl)
	}
	s.data[key] = e
}

func (s *Store) Get(key string) (string, bool) {
	e, ok := s.live(key)
	return e.value, ok
}

func (s *Store) Delete(key string) bool {
	if _, ok := s.live(key); !ok {
		return false
	}
	s.remember(key)
	delete(s.data, key)
	return true
}

func (s *Store) Incr(key string) (int, error) {
	e, ok := s.live(key)
	n := 0
	if ok {
		var err error
		if n, err = strconv.Atoi(e.value); err != nil {
			return 0, fmt.Errorf("incr %s: %w", key, ErrNotInteger)
		}
	}
	s.remember(key)
	e.value = strconv.Itoa(n + 1)
	s.data[key] = e
	return n + 1, nil
}

func (s *Store) TTL(key string) (time.Duration, bool) {
	e, ok := s.live(key)
	if !ok || e.expires.IsZero() {
		return 0, false
	}
	return e.expires.Sub(s.clock.Now()), true
}

func (s *Store) Keys(prefix string) []string {
	keys := []string{}
	for k := range s.data {
		if _, ok := s.live(k); ok && strings.HasPrefix(k, prefix) {
			keys = append(keys, k)
		}
	}
	slices.Sort(keys)
	return keys
}

func (s *Store) Begin()     { s.txs = append(s.txs, map[string]undo{}) }
func (s *Store) Depth() int { return len(s.txs) }

func (s *Store) Rollback() error {
	if len(s.txs) == 0 {
		return ErrNoTransaction
	}
	tx := s.txs[len(s.txs)-1]
	s.txs = s.txs[:len(s.txs)-1]
	for key, u := range tx {
		if u.existed {
			s.data[key] = u.old
		} else {
			delete(s.data, key)
		}
	}
	return nil
}

func (s *Store) Commit() error {
	if len(s.txs) == 0 {
		return ErrNoTransaction
	}
	tx := s.txs[len(s.txs)-1]
	s.txs = s.txs[:len(s.txs)-1]
	if len(s.txs) > 0 { // the outer transaction must be able to undo these changes too
		outer := s.txs[len(s.txs)-1]
		for key, u := range tx {
			if _, seen := outer[key]; !seen {
				outer[key] = u
			}
		}
	}
	return nil
}
