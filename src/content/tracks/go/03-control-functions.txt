## if

```go
package main

import (
	"fmt"
	"strconv"
)

func main() {
	temperature := 31
	if temperature > 30 {
		fmt.Println("hot")
	} else if temperature > 15 {
		fmt.Println("mild")
	} else {
		fmt.Println("cold")
	}

	// an if can start with a short statement; n and err exist only inside the if/else
	if n, err := strconv.Atoi("42"); err != nil {
		fmt.Println("not a number")
	} else {
		fmt.Println("double:", n*2)
	}
}
```

- No parentheses around the condition, and the **braces are required**, even for one line.
- The condition must be a `bool`: `if count {}` doesn't compile; write `if count != 0 {}`.
- The **`if init; condition`** form keeps short-lived variables (typically `err`) out of the rest of the function.

## The early-return style

Go code avoids deep nesting: handle the special or error case first and **return**, so the normal path stays at the left margin ("the happy path is not indented"):

```go
package main

import "fmt"

func shippingCost(weightKg float64, express bool) float64 {
	if weightKg <= 0 {
		return 0
	}
	cost := 4.90
	if weightKg > 2 {
		cost += (weightKg - 2) * 1.50
	}
	if express {
		cost *= 2
	}
	return cost
}

func main() {
	fmt.Printf("%.2f %.2f %.2f\n", shippingCost(1, false), shippingCost(5, false), shippingCost(5, true))
}
```

## switch

```go
package main

import (
	"fmt"
	"time"
)

func main() {
	day := "sat"
	switch day {
	case "sat", "sun": // several values in one case
		fmt.Println("weekend")
	case "fri":
		fmt.Println("almost weekend")
	default:
		fmt.Println("weekday")
	}

	score := 83
	switch { // no value: each case is a condition, like an if/else chain
	case score >= 90:
		fmt.Println("A")
	case score >= 80:
		fmt.Println("B")
	default:
		fmt.Println("C or below")
	}

	switch hour := time.Date(2026, 1, 1, 14, 0, 0, 0, time.UTC).Hour(); {
	case hour < 12:
		fmt.Println("morning")
	case hour < 18:
		fmt.Println("afternoon")
	}
}
```

Unlike C and Java, Go's `switch` **doesn't fall through**: only the matching case runs, so no `break` is needed. If you really want to continue into the next case, write `fallthrough` (rarely used). Cases are tested top to bottom, and case values can be any expression, not just constants.

=== lesson go-loops | Loops: for and range | 60
## One loop keyword

Go has only **`for`**, in three shapes:

```go
package main

import "fmt"

func main() {
	// 1. classic: init; condition; post
	for i := 0; i < 3; i++ {
		fmt.Print(i, " ")
	}
	fmt.Println()

	// 2. condition only: Go's "while"
	n := 1
	for n < 100 {
		n *= 3
	}
	fmt.Println("first power of 3 over 100:", n)

	// 3. no condition: loop forever, until break or return
	attempts := 0
	for {
		attempts++
		if attempts == 4 {
			break
		}
	}
	fmt.Println("attempts:", attempts)

	// range over an integer (Go 1.22+): 0, 1, 2, 3, 4
	for i := range 5 {
		fmt.Print(i*i, " ")
	}
	fmt.Println()
}
```

## range

`for ... range` walks through a collection, giving an index (or key) and a value:

```go
package main

import "fmt"

func main() {
	fruits := []string{"apple", "kiwi", "plum"}
	for i, f := range fruits {
		fmt.Println(i, f)
	}
	for _, f := range fruits { // _ ignores the index
		fmt.Print(len(f), " ")
	}
	fmt.Println()

	ages := map[string]int{"ada": 36}
	for name, age := range ages {
		fmt.Println(name, age)
	}

	for i, r := range "añb" { // strings: byte position and rune
		fmt.Printf("%d:%c ", i, r)
	}
	fmt.Println()
}
```

The value in a `range` loop is a **copy** of the element: changing it doesn't change the slice. To modify elements, assign through the index: `fruits[i] = ...`. Maps are iterated in a **random order** on purpose, so programs don't depend on it (sort the keys when order matters).

## break, continue and labels

`continue` skips to the next iteration; `break` leaves the loop. In nested loops, a **label** says which loop you mean:

```go
package main

import "fmt"

func main() {
	grid := [][]int{{1, 2, 3}, {4, -1, 6}, {7, 8, 9}}
	sum := 0
search:
	for r, row := range grid {
		for c, v := range row {
			if v < 0 {
				fmt.Printf("negative value at row %d, column %d\n", r, c)
				break search // leaves both loops
			}
			sum += v
		}
	}
	fmt.Println("sum before it:", sum)

	for i := range 10 {
		if i%3 != 0 {
			continue
		}
		fmt.Print(i, " ")
	}
	fmt.Println()
}
```

## Declaring functions

```go
package main

import "fmt"

// parameters of the same type can share it: (a, b int)
func add(a, b int) int {
	return a + b
}

// several results: very common in Go
func divmod(a, b int) (int, int) {
	return a / b, a % b
}

// named results act as variables; a bare return returns them
func stats(nums []float64) (minimum, maximum float64) {
	minimum, maximum = nums[0], nums[0]
	for _, n := range nums[1:] {
		minimum = min(minimum, n)
		maximum = max(maximum, n)
	}
	return
}

// variadic: any number of ints, received as a slice
func sum(nums ...int) int {
	total := 0
	for _, n := range nums {
		total += n
	}
	return total
}

func main() {
	fmt.Println(add(2, 3))
	q, r := divmod(17, 5)
	fmt.Println(q, r)
	lo, hi := stats([]float64{3.5, -1, 8})
	fmt.Println(lo, hi)
	fmt.Println(sum(), sum(1, 2, 3))
	values := []int{4, 5, 6}
	fmt.Println(sum(values...)) // spread a slice into a variadic call
}
```

Arguments are always **passed by value** (copied). Slices and maps contain references to their data, so a function can change their *elements*; Module 4 explains exactly why. There are no default parameter values and no overloading: one name, one function.

## Functions are values

Functions can be stored in variables, passed to other functions and returned from them. A **function literal** (anonymous function) that uses variables from its surroundings is a **closure**: it keeps those variables alive.

```go
package main

import (
	"fmt"
	"strings"
)

func apply(words []string, f func(string) string) []string {
	out := make([]string, 0, len(words))
	for _, w := range words {
		out = append(out, f(w))
	}
	return out
}

func counter() func() int {
	count := 0
	return func() int {
		count++ // the closure keeps its own count
		return count
	}
}

func main() {
	fmt.Println(apply([]string{"go", "is", "fun"}, strings.ToUpper))
	fmt.Println(apply([]string{"go", "is"}, func(s string) string { return s + "!" }))

	next := counter()
	other := counter()
	fmt.Println(next(), next(), next(), other())
}
```

## Recursion

A function may call itself. Each call gets its own variables; a **base case** stops the recursion:

```go
package main

import "fmt"

func factorial(n int) int {
	if n <= 1 {
		return 1
	}
	return n * factorial(n-1)
}

func fib(n int, memo map[int]int) int {
	if n < 2 {
		return n
	}
	if v, ok := memo[n]; ok {
		return v
	}
	memo[n] = fib(n-1, memo) + fib(n-2, memo)
	return memo[n]
}

func main() {
	fmt.Println(factorial(10), fib(80, map[int]int{}))
}
```

## defer

**`defer`** schedules a call to run when the surrounding function returns, however it returns (normally, early, or by panicking). It's Go's way to guarantee clean-up: closing files, unlocking mutexes, printing a summary. Deferred calls run in **reverse order** (last in, first out), and their arguments are evaluated immediately:

```go
package main

import "fmt"

func process(name string) {
	fmt.Println("open", name)
	defer fmt.Println("close", name) // runs when process returns
	for i := 1; i <= 3; i++ {
		defer fmt.Println("deferred", i)
	}
	fmt.Println("working on", name)
}

func main() {
	process("report.txt")
}
```

package main

func filter(nums []int, keep func(int) bool) []int {
	var out []int
	for _, n := range nums {
		if keep(n) {
			out = append(out, n)
		}
	}
	return out
}

	return func(n int) int { return n * factor }
}

func average(nums ...int) (float64, bool) {
	if len(nums) == 0 {
		return 0, false
	}
	total := 0
	for _, n := range nums {
		total += n
	}
	return float64(total) / float64(len(nums)), true

## The error type

In Go, a function that can fail returns an **`error`** as its last result. `nil` means success; anything else describes what went wrong. The caller checks it immediately:

```go
package main

import (
	"errors"
	"fmt"
	"strconv"
)

func parseAge(text string) (int, error) {
	age, err := strconv.Atoi(text)
	if err != nil {
		return 0, fmt.Errorf("parse age %q: %w", text, err) // add context, wrap the cause
	}
	if age < 0 || age > 150 {
		return 0, errors.New("age out of range")
	}
	return age, nil
}

func main() {
	for _, input := range []string{"36", "abc", "200"} {
		age, err := parseAge(input)
		if err != nil {
			fmt.Println("error:", err)
			continue
		}
		fmt.Println("age:", age)
	}
}
```

- **`errors.New("message")`** makes a simple error; **`fmt.Errorf`** formats one.
- The verb **`%w`** **wraps** another error: the message includes it, and the original can still be found later.
- Error messages are lower-case without a final period, because they're often joined: `parse age "abc": strconv.Atoi: parsing "abc": invalid syntax`.
- `if err != nil { return ..., err }` appears constantly in Go code. It's verbose, but every failure point is visible, and each layer can add context.

## Checking which error happened

**Sentinel errors** are package-level error values that callers compare against with **`errors.Is`**, which also looks inside wrapped errors. **`errors.As`** finds an error of a specific *type* in the chain (custom error types arrive with structs and methods in Modules 4 and 5):

```go
package main

import (
	"errors"
	"fmt"
	"io/fs"
	"os"
)

var ErrNotFound = errors.New("not found")
var ErrNoStock = errors.New("out of stock")

var stock = map[string]int{"pen": 3, "lamp": 0}

func reserve(item string) error {
	n, ok := stock[item]
	if !ok {
		return fmt.Errorf("reserve %s: %w", item, ErrNotFound)
	}
	if n == 0 {
		return fmt.Errorf("reserve %s: %w", item, ErrNoStock)
	}
	stock[item] = n - 1
	return nil
}

func main() {
	for _, item := range []string{"pen", "lamp", "desk"} {
		err := reserve(item)
		switch {
		case err == nil:
			fmt.Println("reserved", item)
		case errors.Is(err, ErrNoStock):
			fmt.Println("try again later:", err)
		case errors.Is(err, ErrNotFound):
			fmt.Println("unknown item:", err)
		}
	}

	_, err := os.Open("missing-file.txt")
	var pathErr *fs.PathError
	if errors.As(err, &pathErr) {
		fmt.Println("operation:", pathErr.Op, "| path:", pathErr.Path, "| is not-exist:", errors.Is(err, fs.ErrNotExist))
	}
}
```

Never compare error **messages** as strings: they change. Compare with `errors.Is` and `errors.As`.

## panic and recover

**`panic`** stops normal execution: deferred calls run, then the program crashes with a stack trace. Use it only for **bugs and impossible situations** (an index that can't be out of range, a broken invariant), never for expected failures like bad input or missing files.

**`recover`**, called inside a deferred function, stops a panic and returns its value. Servers use it so that one bad request doesn't bring down the whole process; ordinary code almost never needs it:

```go
package main

import "fmt"

func safeDivide(a, b int) (result int, err error) {
	defer func() {
		if r := recover(); r != nil {
			err = fmt.Errorf("recovered: %v", r)
		}
	}()
	return a / b, nil
}

func main() {
	fmt.Println(safeDivide(10, 2))
	fmt.Println(safeDivide(1, 0))
}
```

Here the deferred closure sets the **named result** `err`, which is how a recovered panic becomes an ordinary error.

package main

import (
	"errors"
	"fmt"
	"strconv"
	"strings"
)

var ErrEmptyKey = errors.New("empty key")

func parseSetting(line string) (string, int, error) {
	key, value, found := strings.Cut(line, "=")
	if !found {
		return "", 0, fmt.Errorf("missing \"=\" in %q", line)
	}
	if key == "" {
		return "", 0, fmt.Errorf("line %q: %w", line, ErrEmptyKey)
	}
	n, err := strconv.Atoi(value)
	if err != nil {
		return "", 0, fmt.Errorf("setting %s: %w", key, err)
	}
	return key, n, nil
}

func main() {
	empty := 0
	for _, line := range []string{"timeout=30", "verbose", "=5", "retries=three", "port=8080"} {
		key, value, err := parseSetting(line)
		if err != nil {
			fmt.Println("error:", err)
			if errors.Is(err, ErrEmptyKey) {
				empty++
			}
			var numErr *strconv.NumError
			if errors.As(err, &numErr) {
				fmt.Printf("  bad number: %q\n", numErr.Num)
			}
			continue
		}
		fmt.Println(key, "=", value)
	}
	fmt.Println("empty keys:", empty)
}

package main

import (
	"bufio"
	"errors"
	"fmt"
	"os"
	"strconv"
	"strings"
)

var ErrDivByZero = errors.New("division by zero")

func eval(expr string) (float64, error) {
	f := strings.Fields(expr)
	if len(f) != 3 {
		return 0, fmt.Errorf("want \"a op b\", got %q", expr)
	}
	a, err := number(f[0])
	if err != nil {
		return 0, err
	}
	b, err := number(f[2])
	if err != nil {
		return 0, err
	}
	switch f[1] {
	case "+":
		return a + b, nil
	case "-":
		return a - b, nil
	case "*":
		return a * b, nil
	case "/":
		if b == 0 {
			return 0, ErrDivByZero
		}
		return a / b, nil
	}
	return 0, fmt.Errorf("unknown operator %q", f[1])
}

func number(text string) (float64, error) {
	v, err := strconv.ParseFloat(text, 64)
	if err != nil {
		return 0, fmt.Errorf("bad operand %q: %w", text, err)
	}
	return v, nil
}

func main() {
	in := bufio.NewScanner(os.Stdin)
	zeros := 0
	for in.Scan() {
		expr := in.Text()
		v, err := eval(expr)
		if err != nil {
			fmt.Println("error:", err)
			if errors.Is(err, ErrDivByZero) {
				zeros++
			}
			continue
		}
		fmt.Printf("%s = %g\n", expr, v)
	}
	fmt.Println("divisions by zero:", zeros)
}
